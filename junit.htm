/**
 * Batch skeleton generator:
 * - SUT arranged via `new` on a usable public ctor (TG-NOINSTSUT); Instancio only for the
 *   documented fallbacks (no/self-referencing public ctor, ctor fires project statics)
 * - Instancio for concrete collaborator/value args (unchanged v3.4 behaviour)
 * - Mockito.mock for interfaces; minimal stubbing for smoke act
 * - instance + static in same class; static-only SUTs get static tests only
 * - SUT contract methods: declared; if empty/thin → interfaces + immediate parent
 * - mirrors declared checked exceptions on generated @Test methods
 * - skip if *{suffix} already exists
 *
 * Array TypeName via {@link TypeName#get(java.lang.reflect.Type)} (javaPoetType).
 */
public final class TestSkeletonGenerator {

    private static final ClassName INSTANCIO = ClassName.get("org.instancio", "Instancio");
    private static final ClassName SELECT = ClassName.get("org.instancio", "Select");
    private static final ClassName MOCKITO = ClassName.get("org.mockito", "Mockito");
    private static final ClassName MATCHERS = ClassName.get("org.mockito", "ArgumentMatchers");
    private static final ClassName TEST = ClassName.get("org.junit.jupiter.api", "Test");
    private static final ClassName BEFORE_EACH = ClassName.get("org.junit.jupiter.api", "BeforeEach");
    private static final ClassName AFTER_EACH = ClassName.get("org.junit.jupiter.api", "AfterEach");
    private static final ClassName DISABLED = ClassName.get("org.junit.jupiter.api", "Disabled");
    private static final ClassName ASSERTIONS = ClassName.get("org.junit.jupiter.api", "Assertions");
    private static final ClassName MOCKED_STATIC = ClassName.get("org.mockito", "MockedStatic");
    private static final ClassName MOCKED_CONSTRUCTION = ClassName.get("org.mockito", "MockedConstruction");

    private TestSkeletonGenerator() {}

    public static void run() throws Exception {
        Path testSourceRoot = Config.testSourceRoot();
        String suffix = Config.TEST_NAME_SUFFIX;
        List<String> classes = Config.readClasses();

        for (String fqcn : classes) {
            try {
                writeSkeleton(Class.forName(fqcn, false, TestSkeletonGenerator.class.getClassLoader()),
                        testSourceRoot, suffix);
            } catch (Throwable t) {
                String why = (t.getMessage() != null) ? t.getMessage() : t.toString();
                System.out.println("SKIP (unresolved " + t.getClass().getSimpleName() + ": " + why + "): " + fqcn);
            }
        }
    }

    static void writeSkeleton(Class<?> sut, Path testSourceRoot, String suffix) throws Exception {
        String testPackage = sut.getPackageName();
        String testSimpleName = sut.getSimpleName() + suffix;
        Path target = testSourceRoot
                .resolve(testPackage.replace('.', '/'))
                .resolve(testSimpleName + ".java");
        if (Files.exists(target)) {
            System.out.println("SKIP (exists): " + target);
            return;
        }

        boolean staticOnly = isStaticOnlyClass(sut);
        // BUG-2a (agent5 merge): an enum with an abstract method reports isAbstract()==true but IS
        // instantiable via its constants (Instancio yields a constant) — don't skip its instance methods.
        boolean abstractSut = !sut.isInterface() && !sut.isEnum()
                && java.lang.reflect.Modifier.isAbstract(sut.getModifiers());
        boolean genericSut = sut.getTypeParameters().length > 0;   // Instancio can't build Foo<T> without a type witness
        boolean ifaceSut = sut.isInterface();
        boolean uninstantiable = abstractSut || genericSut || ifaceSut;
        if (uninstantiable) {
            System.out.println("WARN " + (ifaceSut ? "interface" : genericSut ? "generic" : "abstract")
                    + " SUT — instance tests skipped (cannot instantiate reliably): " + sut.getName());
        }
        List<Method> instanceMethods =
                (staticOnly || uninstantiable) ? List.of() : contractInstanceMethods(sut);
        List<Method> staticMethods = contractStaticMethods(sut);
        boolean staticPrefix = !staticOnly && !instanceMethods.isEmpty();

        TypeSpec.Builder testClass = TypeSpec.classBuilder(testSimpleName)
                .addModifiers(Modifier.PUBLIC)
                .addJavadoc(
                        "AUTO-GENERATED arrange skeleton for $L.\n"
                                + "Instancio SUT + locals per test; real assertions to be filled in after generation.\n",
                        sut.getName());

        // TG-BOILERDET: facades the SUT-set consumes get deterministic delegate-swap fixtures;
        // no facade detected -> every path below is byte-identical to the v3.4 output.
        List<SMCFixtureScanner.Facade> facades = SMCFixtureScanner.detect(sut);
        Set<Class<?>> facadeSuppress = SMCFixtureScanner.facadeClasses(facades);
        if (!facades.isEmpty()) {
            for (SMCFixtureScanner.Facade facade : facades) {
                ClassName fixtureName;
                if ("NESTED".equals(Config.FIXTURE_PLACEMENT)) {
                    // TWEAK-POINT (FIXTURE_PLACEMENT=NESTED): same fixture, nested inside this test class.
                    testClass.addType(FixtureGenerator.nestedType(facade));
                    fixtureName = ClassName.get(testPackage, testSimpleName,
                            FixtureGenerator.fixtureSimpleName(facade));
                } else {
                    fixtureName = FixtureGenerator.write(facade, testSourceRoot);
                }
                testClass.addField(FieldSpec.builder(fixtureName, fixtureFieldName(facade),
                                Modifier.PRIVATE, Modifier.FINAL)
                        .initializer("new $T()", fixtureName)
                        .build());
            }
        }

        if (!instanceMethods.isEmpty()) {
            testClass.addField(FieldSpec.builder(ClassName.get(sut), "sut", Modifier.PRIVATE).build());
        }
        if (!facades.isEmpty()) {
            testClass.addMethod(boilerplateSetup(sut, facades, !instanceMethods.isEmpty()));
            testClass.addMethod(boilerplateTearDown(facades));
        } else if (!instanceMethods.isEmpty()) {
            testClass.addMethod(setUpMethod(sut));
        }
        if (!instanceMethods.isEmpty()) {
            Set<String> usedNames = new HashSet<>();
            for (Method m : instanceMethods) {
                testClass.addMethod(instanceTest(sut, m, usedNames, facadeSuppress));
            }
        }
        if (!staticMethods.isEmpty()) {
            Set<String> usedNames = instanceMethods.stream()
                    .map(TestSkeletonGenerator::safeName)
                    .collect(java.util.stream.Collectors.toCollection(HashSet::new));
            for (Method m : staticMethods) {
                testClass.addMethod(staticTest(sut, m, staticPrefix, usedNames, facadeSuppress));
            }
        }
        if (instanceMethods.isEmpty() && staticMethods.isEmpty()) {
            System.out.println("WARN no public methods: " + sut.getName());
            testClass.addJavadoc("No public methods discovered.\n");
        }

        JavaFile.builder(testPackage, testClass.build())
                .addFileComment("AUTO-GENERATED by testgen TestSkeletonGenerator")
                .build()
                .writeTo(testSourceRoot);

        System.out.println("Wrote " + testPackage + "." + testSimpleName
                + " (instance=" + instanceMethods.size() + ", static=" + staticMethods.size() + ")");
    }

    private static MethodSpec setUpMethod(Class<?> sut) {
        MethodSpec.Builder b = MethodSpec.methodBuilder("setUp")
                .addAnnotation(BEFORE_EACH)
                .addModifiers(Modifier.PUBLIC)
                .addException(Exception.class);   // stubbed collaborator methods may throw checked exceptions
        arrangeInto(b, sut);
        return b.build();
    }

    /**
     * TG-NOINSTSUT: the public constructor chosen for arranging the SUT via {@code new}, or null
     * when Instancio must stay (no public ctor / self-referencing param / ctor-or-field-initializer
     * fires project statics that {@code new} would execute for real).
     * Choice among usable ctors: max parameter count (DI-shaped), tie-broken deterministically on
     * the parameter-type signature string.
     */
    private static java.lang.reflect.Constructor<?> noInstSutCtor(Class<?> sut) {
        List<java.lang.reflect.Constructor<?>> usable = new ArrayList<>();
        for (java.lang.reflect.Constructor<?> c : sut.getConstructors()) {
            if (!ctorSelfReferencing(sut, c) && !ctorFiresProjectStatics(sut, c)) {
                usable.add(c);
            }
        }
        if (usable.isEmpty()) {
            return null;
        }
        usable.sort(java.util.Comparator
                .comparingInt((java.lang.reflect.Constructor<?> c) -> c.getParameterCount()).reversed()
                .thenComparing(TestSkeletonGenerator::ctorSignature));
        return usable.get(0);
    }

    /** A ctor taking the SUT's own type cannot be arranged without recursing — Instancio stays. */
    private static boolean ctorSelfReferencing(Class<?> sut, java.lang.reflect.Constructor<?> c) {
        for (Class<?> pt : c.getParameterTypes()) {
            if (pt.equals(sut)) {
                return true;
            }
        }
        return false;
    }

    /** Statics scan of ONE class's ctors (paramCount -1 = all) + instance-field initializers. */
    private static boolean classFiresProjectStatics(Class<?> owner, Class<?> sut, int paramCount) {
        try {
            Path src = Config.mainSourceRoot().resolve(owner.getName().replace('.', '/') + ".java");
            if (!Files.isRegularFile(src)) {
                return false;
            }
            CompilationUnit cu = StaticJavaParser.parse(src);
            String pkg = owner.getPackageName();

            List<com.github.javaparser.ast.Node> scopes = new ArrayList<>();
            for (com.github.javaparser.ast.body.ConstructorDeclaration cd
                    : cu.findAll(com.github.javaparser.ast.body.ConstructorDeclaration.class)) {
                if (cd.getNameAsString().equals(owner.getSimpleName())
                        && (paramCount < 0 || cd.getParameters().size() == paramCount)) {
                    scopes.add(cd);
                }
            }
            for (com.github.javaparser.ast.body.FieldDeclaration fd
                    : cu.findAll(com.github.javaparser.ast.body.FieldDeclaration.class)) {
                if (!fd.isStatic()) {
                    scopes.add(fd);   // instance-field initializers also run on `new`
                }
            }
            for (com.github.javaparser.ast.body.InitializerDeclaration id
                    : cu.findAll(com.github.javaparser.ast.body.InitializerDeclaration.class)) {
                if (!id.isStatic()) {
                    scopes.add(id);   // instance initializer blocks also run on `new` (a11 break #3)
                }
            }

            // Unscoped calls in a ctor/init are the SUT's own methods; their bodies run on `new`
            // too, so walk them one level (a11 break #2). Depth 1 over-retains; it never misses
            // the direct case and the differential backstops deeper chains.
            List<com.github.javaparser.ast.Node> expanded = new ArrayList<>(scopes);
            for (com.github.javaparser.ast.Node scope : scopes) {
                for (MethodCallExpr call : scope.findAll(MethodCallExpr.class)) {
                    if (call.getScope().isEmpty()) {
                        for (com.github.javaparser.ast.body.MethodDeclaration md
                                : cu.findAll(com.github.javaparser.ast.body.MethodDeclaration.class)) {
                            if (md.getNameAsString().equals(call.getNameAsString())) {
                                expanded.add(md);
                            }
                        }
                    }
                }
            }
            scopes = expanded;

            for (com.github.javaparser.ast.Node scope : scopes) {
                // `new ProjectType()` runs that type's ctor for real (a11 break #1): retain.
                for (ObjectCreationExpr oce : scope.findAll(ObjectCreationExpr.class)) {
                    String typeName = oce.getType().getNameAsString();
                    Class<?> created = typeName.contains(".") ? tryLoad(typeName)
                            : resolveSimple(cu, pkg, typeName);
                    if (created != null && isProjectType(created) && !created.equals(sut)) {
                        return true;
                    }
                }
                for (MethodCallExpr call : scope.findAll(MethodCallExpr.class)) {
                    if (call.getScope().isPresent()) {
                        com.github.javaparser.ast.expr.Expression s = call.getScope().get();
                        String q = null;
                        if (s instanceof NameExpr ne && !ne.getNameAsString().isEmpty()
                                && Character.isUpperCase(ne.getNameAsString().charAt(0))) {
                            q = ne.getNameAsString();
                        } else if (s instanceof com.github.javaparser.ast.expr.FieldAccessExpr fae) {
                            q = fae.toString();
                        }
                        if (q != null) {
                            Class<?> c = q.contains(".") ? tryLoad(q) : resolveSimple(cu, pkg, q);
                            if (c != null && isProjectType(c) && !c.equals(sut)) {
                                return true;
                            }
                        }
                    }
                }
            }
            return false;
        } catch (Throwable t) {
            return false;   // unparseable source ⇒ no evidence of statics; differential is the backstop
        }
    }

    /** Why Instancio stays for this SUT — emitted into the generated setUp so the reason ships. */
    private static String instancioRetainedReason(Class<?> sut) {
        if (sut.getConstructors().length == 0) {
            return "no public constructor";
        }
        boolean allSelfRef = true;
        for (java.lang.reflect.Constructor<?> c : sut.getConstructors()) {
            if (!ctorSelfReferencing(sut, c)) {
                allSelfRef = false;
                break;
            }
        }
        if (allSelfRef) {
            return "every public constructor takes the SUT's own type (self-referencing)";
        }
        return "constructor or field initializer invokes project statics (new would fire them for real)";
    }

    /** TG-NOINSTSUT: arrange each ctor param with the EXISTING arg machinery, then {@code new} the SUT. */
    private static void ctorArrangeInto(MethodSpec.Builder b, Class<?> sut,
            java.lang.reflect.Constructor<?> ctor) {
        List<MockRef> mocks = new ArrayList<>();
        List<String> args = new ArrayList<>();
        java.lang.reflect.Type[] gpts = ctor.getGenericParameterTypes();
        Parameter[] ps = ctor.getParameters();

        for (int i = 0; i < ps.length; i++) {
            Class<?> ptype = resolveParamType(sut, gpts[i], ps[i].getType());
            String var = "c" + i + "_" + simpleParamName(ps[i]);
            b.addCode(arrangeLine(ptype, var, mocks));
            args.add(var);
        }
        b.addStatement("sut = new $T($L)", ClassName.get(sut), String.join(", ", args));

        for (MockRef mock : mocks) {
            FieldMockRef fm = soleFieldOfType(sut, mock.type(), mock.varName());
            if (fm != null) {   // exactly one matching SUT field ⇒ reuse the richer field-aware stubs
                b.addCode(minimalStubs(mock, fieldBindings(sut, fm)));
                b.addCode(factoryStubs(sut, mock.type(), fm.fieldName(), mock.varName()));
            } else {
                b.addCode(minimalStubs(mock));
            }
        }
    }

    /** The single SUT instance field of exactly this type, or null when zero/ambiguous. */
    private static FieldMockRef soleFieldOfType(Class<?> sut, Class<?> type, String varName) {
        Field match = null;
        for (Field f : instanceFields(sut)) {
            if (f.getType().equals(type)) {
                if (match != null) {
                    return null;   // ambiguous — plain stubs are the safe choice
                }
                match = f;
            }
        }
        return (match == null) ? null
                : new FieldMockRef(type, varName, match.getDeclaringClass(), match.getName(), false);
    }

    /**
     * True when the ctor body or an instance-field initializer statically calls a project class —
     * {@code new} would fire those statics for real (Instancio bypasses construction), so the
     * Instancio arrangement is kept for safety. The SUPERCLASS chain is walked too (super-ctors
     * and parent field initializers also run on {@code new}); all parent ctors are scanned, which
     * over-retains rather than under-retains. Source unavailable ⇒ false (same policy as
     * {@link #analyzeStaticsAndNew}); the differential run is the backstop.
     */
    private static boolean ctorFiresProjectStatics(Class<?> sut, java.lang.reflect.Constructor<?> ctor) {
        if (classFiresProjectStatics(sut, sut, ctor.getParameterCount())) {
            return true;
        }
        for (Class<?> parent = sut.getSuperclass();
                parent != null && parent != Object.class && isProjectType(parent);
                parent = parent.getSuperclass()) {
            if (classFiresProjectStatics(parent, sut, -1)) {   // -1: every ctor of the parent
                return true;
            }
        }
        return false;
    }

    /** Deterministic tie-break key: the ctor's parameter-type signature string. */
    private static String ctorSignature(java.lang.reflect.Constructor<?> c) {
        StringBuilder sb = new StringBuilder();
        for (Class<?> pt : c.getParameterTypes()) {
            sb.append(pt.getName()).append(',');
        }
        return sb.toString();
    }

    /** Fixture lifecycle: install first in setup, then the standard arrange; restore in tearDown. */
    private static MethodSpec boilerplateSetup(Class<?> sut, List<SMCFixtureScanner.Facade> facades,
            boolean arrangeSut) {
        MethodSpec.Builder b = MethodSpec.methodBuilder("setup")
                .addAnnotation(BEFORE_EACH)
                .addModifiers(Modifier.PUBLIC)
                .addException(Exception.class);
        for (SMCFixtureScanner.Facade facade : facades) {
            b.addStatement("$L.install()", fixtureFieldName(facade));
        }
        if (arrangeSut) {
            arrangeInto(b, sut);
        }
        return b.build();
    }

    /** Restore is mandatory — the swapped delegates are JVM-global state (lab play). */
    private static MethodSpec boilerplateTearDown(List<SMCFixtureScanner.Facade> facades) {
        MethodSpec.Builder b = MethodSpec.methodBuilder("tearDown")
                .addAnnotation(AFTER_EACH)
                .addModifiers(Modifier.PUBLIC)
                .addException(Exception.class);
        for (int i = facades.size() - 1; i >= 0; i--) {
            b.addStatement("$L.restore()", fixtureFieldName(facades.get(i)));
        }
        return b.build();
    }

    private static String fixtureFieldName(SMCFixtureScanner.Facade facade) {
        String simple = FixtureGenerator.fixtureSimpleName(facade);
        return Character.toLowerCase(simple.charAt(0)) + simple.substring(1);
    }

    private static void arrangeInto(MethodSpec.Builder b, Class<?> sut) {
        /*
         * TWEAK-POINT (TG-NOINSTSUT): the SUT itself is arranged via `new` when a public ctor is
         * usable; Instancio only for the documented fallbacks (reason emitted into the test).
         * Ctor choice: noInstSutCtor(). Safety scan: classFiresProjectStatics().
         */
        java.lang.reflect.Constructor<?> ctor = noInstSutCtor(sut);
        if (ctor != null) {
            ctorArrangeInto(b, sut, ctor);
            return;
        }
        b.addComment("Instancio retained for SUT construction: $L", instancioRetainedReason(sut));

        List<FieldMockRef> fieldMocks = new ArrayList<>();
        for (Field field : instanceFields(sut)) {
            Class<?> ft = field.getType();
            Class<?> coll = collectionImpl(ft);
            Class<?> sealedImpl = sealedConcrete(ft);   // sealed types are unmockable → use a permitted impl
            boolean abstractCls = !ft.isArray() && !ft.isInterface() && !ft.isPrimitive()
                    && java.lang.reflect.Modifier.isAbstract(ft.getModifiers());   // primitives report isAbstract too
            if (coll == null && sealedImpl == null && !ft.isInterface() && !abstractCls
                    && !(ft.getTypeParameters().length > 0 && !ft.isArray() && ft != Class.class)) {
                continue;   // mock interface/abstract + generic-concrete fields (Instancio can't build them); Class is unmockable
            }
            boolean realValue = coll != null || sealedImpl != null;   // real object, not a mock → no stubbing
            String var = field.getName() + (realValue ? "Val" : "Mock");
            fieldMocks.add(new FieldMockRef(ft, var, field.getDeclaringClass(), field.getName(), realValue));
            if (coll != null) {   // real empty collection, not a mock (mock iterator() == null → NPE when iterated)
                b.addStatement("$T $L = new $T<>()", ClassName.get(ft), var, ClassName.get(coll));
            } else if (sealedImpl != null) {   // permitted concrete impl of the sealed type
                b.addStatement("$T $L = $T.create($T.class)", ClassName.get(ft), var, INSTANCIO, ClassName.get(sealedImpl));
            } else {
                b.addStatement("$T $L = $T.mock($T.class)",
                        ClassName.get(ft), var, MOCKITO, ClassName.get(ft));
            }
        }

        if (fieldMocks.isEmpty()) {
            b.addStatement("sut = $T.create($T.class)", INSTANCIO, ClassName.get(sut));
        } else {
            CodeBlock.Builder chain = CodeBlock.builder().add("sut = $T.of($T.class)",
                    INSTANCIO, ClassName.get(sut));
            for (FieldMockRef mock : fieldMocks) {
                chain.add("\n    .set($T.field($T.class, $S), $L)",
                        SELECT, ClassName.get(mock.declaringClass), mock.fieldName, mock.varName);
            }
            chain.add("\n    .create();\n");
            b.addCode(chain.build());
        }

        for (FieldMockRef mock : fieldMocks) {
            if (mock.skipStub()) continue;   // real collection/sealed value, not a mock — nothing to stub
            b.addCode(minimalStubs(new MockRef(mock.type(), mock.varName()), fieldBindings(sut, mock)));
            b.addCode(factoryStubs(sut, mock.type(), mock.fieldName(), mock.varName()));   // call-site Class<T> binds
        }
    }

    private static MethodSpec instanceTest(Class<?> sut, Method method, Set<String> usedNames,
            Set<Class<?>> facadeSuppress) {
        MethodSpec.Builder b = MethodSpec.methodBuilder(uniqueName(safeName(method), usedNames))
                .addModifiers(Modifier.PUBLIC)
                .addAnnotation(TEST)
                .addException(Exception.class);   // stubbed collaborator methods may throw checked exceptions
        addDeclaredExceptions(method, b);

        String disable = disableReason(method);
        if (disable != null) return disabledTest(b, disable);

        SutStatics st = suppressFacades(analyzeStaticsAndNew(sut, method), facadeSuppress);
        boolean scope = st != null && (!st.staticTargets().isEmpty() || !st.constructedTypes().isEmpty());
        if (scope) {
            b.beginControlFlow("try ($L)", mockScopeResources(st));
            emitStaticStubs(b, st, sut);
        }

        List<MockRef> mocks = new ArrayList<>();
        List<String> methodArgNames = new ArrayList<>();
        java.lang.reflect.Type[] gpts = method.getGenericParameterTypes();
        Parameter[] ps = method.getParameters();
        for (int i = 0; i < ps.length; i++) {
            Class<?> ptype = resolveParamType(sut, gpts[i], ps[i].getType());
            String var = "a" + i + "_" + simpleParamName(ps[i]);
            b.addCode(arrangeLine(ptype, var, mocks));
            methodArgNames.add(var);
        }

        for (MockRef mock : mocks) {
            b.addCode(minimalStubs(mock));
        }

        String callArgs = String.join(", ", methodArgNames);
        if (method.getReturnType() == void.class) {
            b.addStatement("sut.$L($L)", method.getName(), callArgs);
            b.addStatement("$T.assertNotNull(sut)", ASSERTIONS);
        } else if (method.getReturnType().isPrimitive()) {
            b.addStatement("sut.$L($L)", method.getName(), callArgs);
            b.addStatement("$T.assertTrue(true)", ASSERTIONS);
        } else {
            b.addStatement("$T result = sut.$L($L)",
                    TypeName.get(method.getReturnType()), method.getName(), callArgs);
            b.addComment("TODO: assert on result (a null return may be valid — arrange skeleton)");
        }
        if (scope) {
            b.endControlFlow();
        }
        return b.build();
    }

    private static MethodSpec staticTest(Class<?> sut, Method method, boolean prefix, Set<String> usedNames,
            Set<Class<?>> facadeSuppress) {
        String base = prefix ? "static_" + safeName(method) : safeName(method);
        MethodSpec.Builder b = MethodSpec.methodBuilder(uniqueName(base, usedNames))
                .addModifiers(Modifier.PUBLIC)
                .addAnnotation(TEST)
                .addException(Exception.class);   // stubbed collaborator methods may throw checked exceptions
        addDeclaredExceptions(method, b);

        String disable = disableReason(method);
        if (disable != null) return disabledTest(b, disable);

        SutStatics st = suppressFacades(analyzeStaticsAndNew(sut, method), facadeSuppress);
        boolean scope = st != null && (!st.staticTargets().isEmpty() || !st.constructedTypes().isEmpty());
        if (scope) {
            b.beginControlFlow("try ($L)", mockScopeResources(st));
            emitStaticStubs(b, st, sut);
        }

        List<MockRef> mocks = new ArrayList<>();
        List<String> methodArgNames = new ArrayList<>();
        java.lang.reflect.Type[] gpts = method.getGenericParameterTypes();
        Parameter[] ps = method.getParameters();
        for (int i = 0; i < ps.length; i++) {
            Class<?> ptype = resolveParamType(sut, gpts[i], ps[i].getType());
            String var = "a" + i + "_" + simpleParamName(ps[i]);
            b.addCode(arrangeLine(ptype, var, mocks));
            methodArgNames.add(var);
        }
        for (MockRef mock : mocks) {
            b.addCode(minimalStubs(mock));
        }

        String callArgs = String.join(", ", methodArgNames);
        ClassName sutCn = ClassName.get(sut);
        if (method.getReturnType() == void.class) {
            b.addStatement("$T.$L($L)", sutCn, method.getName(), callArgs);
            b.addStatement("$T.assertTrue(true)", ASSERTIONS);
        } else if (method.getReturnType().isPrimitive()) {
            b.addStatement("$T.$L($L)", sutCn, method.getName(), callArgs);
            b.addStatement("$T.assertTrue(true)", ASSERTIONS);
        } else {
            b.addStatement("$T result = $T.$L($L)",
                    TypeName.get(method.getReturnType()), sutCn, method.getName(), callArgs);
            b.addComment("TODO: assert on result (a null return may be valid — arrange skeleton)");
        }
        if (scope) {
            b.endControlFlow();
        }
        return b.build();
    }

    private static List<Field> instanceFields(Class<?> sut) {
        List<Field> fields = new ArrayList<>();
        Class<?> cur = sut;
        while (cur != null && cur != Object.class) {
            // Skip JDK frames (e.g. Throwable.suppressedExceptions) — not SUT collaborators.
            if (isJdkType(cur)) {
                cur = cur.getSuperclass();
                continue;
            }
            for (Field field : cur.getDeclaredFields()) {
                if (!java.lang.reflect.Modifier.isStatic(field.getModifiers())) {
                    fields.add(field);
                }
            }
            cur = cur.getSuperclass();
        }
        return fields;
    }

    private static boolean isJdkType(Class<?> type) {
        Package p = type.getPackage();
        if (p == null) {
            return false;
        }
        String n = p.getName();
        return n.equals("java") || n.startsWith("java.")
                || n.equals("javax") || n.startsWith("javax.")
                || n.equals("jdk") || n.startsWith("jdk.");
    }

    /** True if the class's top-level source exists under the main source root (project type, not JDK/third-party). */
    private static boolean isProjectType(Class<?> type) {
        if (type == null) {
            return false;
        }
        Class<?> top = type;
        while (top.getEnclosingClass() != null) {
            top = top.getEnclosingClass();
        }
        return Files.exists(Config.mainSourceRoot().resolve(top.getName().replace('.', '/') + ".java"));
    }

    /** Resolve a parameter's generic type against the SUT's superclass parameterization, so a type
     *  variable from a generic parent (e.g. T extends ClaimsHolder) becomes the concrete type the
     *  call site requires (e.g. HeaderClaimsHolder) rather than the erased bound. Falls back to erased. */
    /** Honest skip: a compiling @Disabled test with the reason, for methods the skeleton can't arrange. */
    private static MethodSpec disabledTest(MethodSpec.Builder b, String reason) {
        b.addAnnotation(AnnotationSpec.builder(DISABLED).addMember("value", "$S", reason).build());
        b.addStatement("$T.assertTrue(true)", ASSERTIONS);
        return b.build();
    }

    private static Class<?> resolveParamType(Class<?> sut, java.lang.reflect.Type generic, Class<?> erased) {
        try {
            Class<?> multi = multiBoundConcrete(generic);
            if (multi != null) return multi;   // <T extends Number & Comparable<T>> -> Integer (satisfies all)
            Class<?> r = toClass(resolveType(generic, typeBindings(sut)));
            return (r != null && r != Object.class) ? r : erased;
        } catch (Throwable t) {
            return erased;
        }
    }

    /** For a type variable with an INTERSECTION bound whose first bound alone can't be denoted as an
     *  argument (Number isn't Comparable), find a concrete type assignable to ALL bounds so the call
     *  type-checks. Returns null when the first bound already satisfies the rest (normal path) or when
     *  no common concrete exists (caller then @Disables the method). */
    private static Class<?> multiBoundConcrete(java.lang.reflect.Type generic) {
        if (!(generic instanceof java.lang.reflect.TypeVariable<?> tv)) return null;
        java.lang.reflect.Type[] bounds = tv.getBounds();
        if (bounds.length <= 1) return null;
        Class<?> first = toClass(bounds[0]);
        boolean firstSatisfies = true;
        for (int i = 1; i < bounds.length; i++) {
            if (!toClass(bounds[i]).isAssignableFrom(first)) { firstSatisfies = false; break; }
        }
        if (firstSatisfies) return null;   // first bound denotable on its own — normal path handles it
        for (Class<?> c : new Class<?>[]{Integer.class, Long.class, Double.class, String.class, Boolean.class}) {
            boolean ok = true;
            for (java.lang.reflect.Type b : bounds) if (!toClass(b).isAssignableFrom(c)) { ok = false; break; }
            if (ok) return c;
        }
        return null;   // no common concrete -> method gets @Disabled
    }

    /** Non-null reason when a method can't be safely arranged: an intersection-bound type-var param
     *  with no denotable/concrete satisfying type. The test is emitted @Disabled (honest skip) rather
     *  than as code that won't compile. */
    private static String disableReason(Method m) {
        for (java.lang.reflect.Type gt : m.getGenericParameterTypes()) {
            if (gt instanceof java.lang.reflect.TypeVariable<?> tv && tv.getBounds().length > 1) {
                Class<?> first = toClass(tv.getBounds()[0]);
                boolean firstSatisfies = true;
                for (int i = 1; i < tv.getBounds().length; i++) {
                    if (!toClass(tv.getBounds()[i]).isAssignableFrom(first)) { firstSatisfies = false; break; }
                }
                if (!firstSatisfies && multiBoundConcrete(gt) == null) {
                    return "testgen: cannot arrange intersection-bounded type parameter <" + tv.getName() + ">";
                }
            }
        }
        return null;
    }

    private static java.util.Map<java.lang.reflect.TypeVariable<?>, java.lang.reflect.Type> typeBindings(Class<?> sut) {
        java.util.Map<java.lang.reflect.TypeVariable<?>, java.lang.reflect.Type> map = new java.util.HashMap<>();
        collectBindings(sut, map);
        return map;
    }

    /** Collect type-var bindings from the SUT's generic superclasses AND generic interfaces
     *  (e.g. Comparable&lt;Priority&gt; -&gt; T=Priority), recursively. */
    private static void collectBindings(Class<?> c,
            java.util.Map<java.lang.reflect.TypeVariable<?>, java.lang.reflect.Type> map) {
        if (c == null || c == Object.class) {
            return;
        }
        bindArgs(c.getGenericSuperclass(), map);
        collectBindings(c.getSuperclass(), map);
        for (java.lang.reflect.Type it : c.getGenericInterfaces()) {
            bindArgs(it, map);
            if (it instanceof java.lang.reflect.ParameterizedType ipt) {
                collectBindings((Class<?>) ipt.getRawType(), map);
            } else if (it instanceof Class<?> ic) {
                collectBindings(ic, map);
            }
        }
    }

    /** Bindings for a collaborator field: SUT-level bindings plus the field's own type args
     *  (store:Store&lt;String&gt; -> Store's T -> String), so its methods' T returns resolve concretely. */
    private static java.util.Map<java.lang.reflect.TypeVariable<?>, java.lang.reflect.Type> fieldBindings(
            Class<?> sut, FieldMockRef mock) {
        java.util.Map<java.lang.reflect.TypeVariable<?>, java.lang.reflect.Type> map = typeBindings(sut);
        try {
            bindArgs(mock.declaringClass().getDeclaredField(mock.fieldName()).getGenericType(), map);
        } catch (NoSuchFieldException ignored) {
            // field not reflectively found (shouldn't happen) — fall back to SUT-level bindings only
        }
        return map;
    }

    private static void bindArgs(java.lang.reflect.Type t,
            java.util.Map<java.lang.reflect.TypeVariable<?>, java.lang.reflect.Type> map) {
        if (t instanceof java.lang.reflect.ParameterizedType pt) {
            Class<?> raw = (Class<?>) pt.getRawType();
            java.lang.reflect.TypeVariable<?>[] vars = raw.getTypeParameters();
            java.lang.reflect.Type[] args = pt.getActualTypeArguments();
            for (int i = 0; i < vars.length && i < args.length; i++) {
                map.put(vars[i], resolveType(args[i], map));
            }
        }
    }

    private static java.lang.reflect.Type resolveType(java.lang.reflect.Type t,
            java.util.Map<java.lang.reflect.TypeVariable<?>, java.lang.reflect.Type> map) {
        if (t instanceof java.lang.reflect.TypeVariable<?> tv) {
            java.lang.reflect.Type bound = map.get(tv);
            if (bound != null && bound != tv) {
                return resolveType(bound, map);
            }
            java.lang.reflect.Type[] bounds = tv.getBounds();
            return bounds.length > 0 ? bounds[0] : Object.class;
        }
        return t;
    }

    /** A concrete Class for a collaborator method's generic return, resolving type variables (and T[])
     *  via the collaborator's bindings; null if it can't be resolved to a concrete non-Object type. */
    private static Class<?> resolveReturnClass(java.lang.reflect.Type gret,
            java.util.Map<java.lang.reflect.TypeVariable<?>, java.lang.reflect.Type> bindings) {
        if (bindings.isEmpty()) return null;
        if (gret instanceof java.lang.reflect.TypeVariable<?>) {
            Class<?> c = toClass(resolveType(gret, bindings));
            return (c != null && c != Object.class) ? c : null;
        }
        if (gret instanceof java.lang.reflect.GenericArrayType gat
                && gat.getGenericComponentType() instanceof java.lang.reflect.TypeVariable<?> comp) {
            Class<?> cc = toClass(resolveType(comp, bindings));
            if (cc != null && cc != Object.class) {
                return java.lang.reflect.Array.newInstance(cc, 0).getClass();   // e.g. String[].class
            }
        }
        return null;
    }

    /** True if the generic return is a type variable (or an array of one) — i.e. erased to Object/Object[]. */
    private static boolean isTypeVarReturn(java.lang.reflect.Type gret) {
        if (gret instanceof java.lang.reflect.TypeVariable<?>) return true;
        if (gret instanceof java.lang.reflect.GenericArrayType gat) return isTypeVarReturn(gat.getGenericComponentType());
        return false;
    }

    private static Class<?> toClass(java.lang.reflect.Type t) {
        if (t instanceof Class<?> c) {
            return c;
        }
        if (t instanceof java.lang.reflect.ParameterizedType pt) {
            return (Class<?>) pt.getRawType();
        }
        if (t instanceof java.lang.reflect.GenericArrayType) {
            return Object[].class;
        }
        if (t instanceof java.lang.reflect.TypeVariable<?> tv) {
            java.lang.reflect.Type[] b = tv.getBounds();
            return b.length > 0 ? toClass(b[0]) : Object.class;
        }
        return Object.class;
    }

    private static boolean isStaticOnlyClass(Class<?> sut) {
        return filterInstance(sut.getDeclaredMethods()).isEmpty()
                && !filterStatic(sut.getDeclaredMethods()).isEmpty();
    }

    /** Prefer TypeName.get so arrays (and generics) render correctly. */
    private static TypeName javaPoetType(Class<?> type) {
        return TypeName.get(type);
    }

    /** Concrete impl for a JDK collection/map interface, else null. A mock collection returns
     *  null from iterator()/entrySet() → SUTs that iterate the arg NPE; a real empty one runs. */
    private static Class<?> collectionImpl(Class<?> t) {
        if (!t.isInterface()) return null;
        if (t == java.util.NavigableMap.class || t == java.util.SortedMap.class) return java.util.TreeMap.class;
        if (t == java.util.Map.class) return java.util.HashMap.class;
        if (t == java.util.NavigableSet.class || t == java.util.SortedSet.class) return java.util.TreeSet.class;
        if (t == java.util.Set.class) return java.util.HashSet.class;
        if (t == java.util.Queue.class || t == java.util.Deque.class) return java.util.ArrayDeque.class;
        if (t == java.util.List.class || t == java.util.Collection.class || t == Iterable.class)
            return java.util.ArrayList.class;
        return null;
    }

    /** A permitted, instantiable concrete subtype of a sealed type (recursing through sealed subtiers),
     *  else null. Sealed types are unmockable (Mockito refuses) — build a permitted impl instead. */
    private static Class<?> sealedConcrete(Class<?> t) {
        if (t == null || !t.isSealed()) return null;
        for (Class<?> sub : t.getPermittedSubclasses()) {
            if (sub.isSealed()) {
                Class<?> deeper = sealedConcrete(sub);
                if (deeper != null) return deeper;
            } else if (!sub.isInterface() && !java.lang.reflect.Modifier.isAbstract(sub.getModifiers())) {
                return sub;   // concrete record/class permitted by the seal
            }
        }
        return null;
    }

    private static CodeBlock arrangeLine(Class<?> type, String varName, List<MockRef> mocks) {
        Class<?> sealedImpl = sealedConcrete(type);
        if (sealedImpl != null) {   // sealed param -> a permitted concrete impl (sealed types are unmockable)
            return arrangeConcrete(sealedImpl, varName, mocks);
        }
        Class<?> coll = collectionImpl(type);
        if (coll != null) {   // real empty collection, not a mock (mock iterator() == null → NPE)
            return CodeBlock.of("$T $L = new $T<>();\n", ClassName.get(type), varName, ClassName.get(coll));
        }
        if (type.isInterface() || (!type.isArray() && !type.isPrimitive() && type != Class.class
                && (java.lang.reflect.Modifier.isAbstract(type.getModifiers())
                        || type.getTypeParameters().length > 0))) {   // generic concrete: Instancio can't build raw
            mocks.add(new MockRef(type, varName));
            return CodeBlock.of("$T $L = $T.mock($T.class);\n",
                    ClassName.get(type), varName, MOCKITO, ClassName.get(type));
        }
        if (type == Class.class) {
            return CodeBlock.of("$T<?> $L = $T.class;\n",
                    ClassName.get(Class.class), varName, ClassName.get(String.class));
        }
        if (type == java.math.BigDecimal.class) {
            return CodeBlock.of("$T $L = $T.ZERO;\n",
                    ClassName.get(java.math.BigDecimal.class), varName,
                    ClassName.get(java.math.BigDecimal.class));
        }
        if (type.isPrimitive()) {
            Class<?> boxed = box(type);
            return CodeBlock.of("$T $L = $T.create($T.class);\n",
                    ClassName.get(boxed), varName, INSTANCIO, ClassName.get(boxed));
        }
        return arrangeConcrete(type, varName, mocks);
    }

    private static CodeBlock arrangeConcrete(Class<?> type, String varName, List<MockRef> mocks) {
        List<Field> ifaceFields = interfaceInstanceFields(type);
        if (ifaceFields.isEmpty()) {
            TypeName tn = javaPoetType(type);
            return CodeBlock.of("$T $L = $T.create($T.class);\n", tn, varName, INSTANCIO, tn);
        }
        CodeBlock.Builder block = CodeBlock.builder();
        for (Field field : ifaceFields) {
            String mockVar = varName + "_" + field.getName() + "Mock";
            mocks.add(new MockRef(field.getType(), mockVar));
            block.addStatement("$T $L = $T.mock($T.class)",
                    ClassName.get(field.getType()), mockVar, MOCKITO, ClassName.get(field.getType()));
        }
        CodeBlock.Builder chain = CodeBlock.builder().add("$T $L = $T.of($T.class)",
                ClassName.get(type), varName, INSTANCIO, ClassName.get(type));
        for (Field field : ifaceFields) {
            String mockVar = varName + "_" + field.getName() + "Mock";
            chain.add("\n    .set($T.field($T.class, $S), $L)",
                    SELECT, ClassName.get(field.getDeclaringClass()), field.getName(), mockVar);
        }
        chain.add("\n    .create();\n");
        block.add(chain.build());
        return block.build();
    }

    private static List<Field> interfaceInstanceFields(Class<?> type) {
        List<Field> out = new ArrayList<>();
        for (Field field : instanceFields(type)) {
            Class<?> ft = field.getType();
            boolean abstractCls = !ft.isArray() && !ft.isInterface() && !ft.isPrimitive()
                    && java.lang.reflect.Modifier.isAbstract(ft.getModifiers());   // primitives report isAbstract too
            if (ft.isInterface() || abstractCls
                    || (ft.getTypeParameters().length > 0 && !ft.isArray() && ft != Class.class)) {
                out.add(field);
            }
        }
        return out;
    }

    private static void addDeclaredExceptions(Method method, MethodSpec.Builder testMethod) {
        for (Class<?> ex : method.getExceptionTypes()) {
            testMethod.addException(TypeName.get(ex));
        }
    }

    private static CodeBlock minimalStubs(MockRef mock) {
        return minimalStubs(mock, java.util.Map.of());
    }

    /** Call-site-aware stubs for a collaborator's {@code <T> T m(Class<T>)} factory methods. The SUT
     *  passes a concrete {@code X.class} that the generic any()-matcher stub can't bind to T (so #37
     *  returns null -> NPE if the SUT derefs). Here we read the SUT source, find {@code field.m(X.class)}
     *  calls, and emit {@code when(mock.m(X.class)).thenReturn(<value of X>)} so T resolves per call. */
    private static CodeBlock factoryStubs(Class<?> sut, Class<?> collab, String fieldName, String mockVar) {
        CodeBlock.Builder out = CodeBlock.builder();
        try {
            Path src = Config.mainSourceRoot().resolve(sut.getName().replace('.', '/') + ".java");
            if (!Files.isRegularFile(src)) return out.build();
            CompilationUnit cu = StaticJavaParser.parse(src);
            String pkg = sut.getPackageName();
            java.util.Set<String> seen = new java.util.HashSet<>();
            for (MethodCallExpr call : cu.findAll(MethodCallExpr.class)) {
                if (call.getScope().isEmpty()) continue;
                com.github.javaparser.ast.expr.Expression scope = call.getScope().get();
                String scopeName = scope instanceof NameExpr ne ? ne.getNameAsString()
                        : scope instanceof com.github.javaparser.ast.expr.FieldAccessExpr fae ? fae.getNameAsString()
                        : null;
                if (!fieldName.equals(scopeName)) continue;
                int[] pos = {-1};
                Class<?> x = classArg(call, cu, pkg, pos);   // the single X.class arg + its position
                if (x == null) continue;
                Method fm = factoryMethod(collab, call.getNameAsString(), call.getArguments().size(), pos[0]);
                if (fm == null || !seen.add(call.getNameAsString() + "|" + pos[0] + "|" + x.getName())) continue;
                out.addStatement("$T.when($L.$L($L)).thenReturn($L)",
                        MOCKITO, mockVar, call.getNameAsString(), factoryMatchers(fm, pos[0], x), stubValue(x));
            }
        } catch (Throwable ignored) {
            // best-effort: on any parse/resolve failure fall back to the generic stub already emitted
        }
        return out.build();
    }

    /** If the call has exactly one {@code X.class} argument, resolve X and record its position; else null. */
    private static Class<?> classArg(MethodCallExpr call, CompilationUnit cu, String pkg, int[] posOut) {
        Class<?> x = null;
        for (int i = 0; i < call.getArguments().size(); i++) {
            if (call.getArgument(i) instanceof com.github.javaparser.ast.expr.ClassExpr ce) {
                if (posOut[0] >= 0) return null;   // more than one Class arg — ambiguous, skip
                posOut[0] = i;
                x = resolveSimple(cu, pkg, ce.getType().asString());
            }
        }
        return x;
    }

    /** Matchers for a factory call: eq(X.class) at the Class position, typed any() elsewhere. */
    private static CodeBlock factoryMatchers(Method fm, int classPos, Class<?> x) {
        Class<?>[] ps = fm.getParameterTypes();
        CodeBlock.Builder mb = CodeBlock.builder();
        for (int i = 0; i < ps.length; i++) {
            if (i > 0) mb.add(", ");
            if (i == classPos) mb.add("$T.eq($T.class)", MATCHERS, ClassName.get(x));
            else mb.add(typedAnyMatcher(ps[i]));
        }
        return mb.build();
    }

    /** A {@code <T> T m(..., Class<T>, ...)} factory method: matching name/arity, type-variable return,
     *  and a {@code Class} parameter at the position the SUT passed {@code X.class}. */
    private static Method factoryMethod(Class<?> collab, String name, int argc, int classPos) {
        for (Method m : collab.getMethods()) {
            if (m.getName().equals(name) && m.getParameterCount() == argc
                    && classPos >= 0 && classPos < m.getParameterCount()
                    && m.getParameterTypes()[classPos] == Class.class
                    && m.getGenericReturnType() instanceof java.lang.reflect.TypeVariable) {
                return m;
            }
        }
        return null;
    }

    /** bindings: type-variable -> concrete type, derived from the collaborator reference's parameterized
     *  type (e.g. field store:Store&lt;String&gt;). Lets a T-returning collaborator method stub the bound
     *  concrete (String) instead of erased Object -> avoids ClassCastException at the SUT call site. */
    private static CodeBlock minimalStubs(MockRef mock,
            java.util.Map<java.lang.reflect.TypeVariable<?>, java.lang.reflect.Type> bindings) {
        CodeBlock.Builder stubs = CodeBlock.builder();
        for (Method m : mock.type.getMethods()) {
            if (!include(m) || java.lang.reflect.Modifier.isStatic(m.getModifiers())) {
                continue;
            }
            if (!Config.STUB_THIRD_PARTY && !isProjectType(m.getDeclaringClass())) {
                continue;   // internal-only: mock third-party collaborators but don't stub their methods
            }
            CodeBlock anyArgs = anyMatchers(m.getParameterTypes());
            java.lang.reflect.Type gret = m.getGenericReturnType();
            Class<?> ret = m.getReturnType();
            Class<?> resolved = resolveReturnClass(gret, bindings);   // bind type-var / T[] via collaborator generics
            if (resolved != null) ret = resolved;
            boolean unresolvedTypeVar = isTypeVarReturn(gret) && (resolved == null);
            if (ret == void.class) {
                stubs.addStatement("$T.doNothing().when($L).$L($L)",
                        MOCKITO, mock.varName, m.getName(), anyArgs);
            } else if (ret.isPrimitive()) {
                stubs.addStatement("$T.doReturn($L).when($L).$L($L)",
                        MOCKITO, primitiveDefaultExpr(ret), mock.varName, m.getName(), anyArgs);
            } else if (unresolvedTypeVar) {   // T bound by the call arg (unknowable here): a bare Object cast
                stubs.addStatement("$T.doReturn(null).when($L).$L($L)",   // to the SUT's expected type would CCE; null casts to anything
                        MOCKITO, mock.varName, m.getName(), anyArgs);
            } else if (ret != Object.class && ret.isAssignableFrom(mock.type)) {   // fluent/self-returning: same mock
                stubs.addStatement("$T.doReturn($L).when($L).$L($L)",   // so chained builder calls resolve (RETURNS_SELF)
                        MOCKITO, mock.varName, mock.varName, m.getName(), anyArgs);
            } else {
                // doReturn(...).when(...) takes Object -> sidesteps the thenReturn compile-time
                // return-type check that fails on erased generic/covariant returns.
                CodeBlock rv = classReturnStub(gret);   // Class<...> -> X.class (Instancio can't build Class)
                if (rv == null) rv = optionalReturnStub(gret);   // real present Optional, not a mock
                if (rv == null) rv = entryReturnStub(gret);   // real Map.entry, not a mock
                if (rv == null) rv = futureReturnStub(gret);   // real completed future, not a mock
                if (rv == null) rv = stubValue(ret);
                stubs.addStatement("$T.doReturn($L).when($L).$L($L)",
                        MOCKITO, rv, mock.varName, m.getName(), anyArgs);
            }
        }
        return stubs.build();
    }

    /** A real completed future for a method returning CompletableFuture/Future/CompletionStage (a mocked
     *  one returns null from join()/get() -> NPE); uses the element type, else completes with null. */
    private static CodeBlock futureReturnStub(java.lang.reflect.Type gret) {
        Class<?> raw = toClass(gret);
        if (raw != java.util.concurrent.CompletableFuture.class && raw != java.util.concurrent.Future.class
                && raw != java.util.concurrent.CompletionStage.class) {
            return null;
        }
        ClassName cf = ClassName.get(java.util.concurrent.CompletableFuture.class);
        if (gret instanceof java.lang.reflect.ParameterizedType pt) {
            Class<?> elem = toClass(pt.getActualTypeArguments()[0]);
            if (elem != null && elem != Object.class) {
                return CodeBlock.of("$T.completedFuture($L)", cf, stubValue(elem));
            }
        }
        return CodeBlock.of("$T.completedFuture(null)", cf);
    }

    /** A real {@code Map.entry(k, v)} for a method returning {@code Map.Entry<K,V>} (a mocked entry
     *  returns null from getKey()/getValue() -> NPE if derefed); null when not a Map.Entry return. */
    private static CodeBlock entryReturnStub(java.lang.reflect.Type gret) {
        if (toClass(gret) != java.util.Map.Entry.class) return null;
        ClassName map = ClassName.get(java.util.Map.class);
        if (gret instanceof java.lang.reflect.ParameterizedType pt
                && pt.getActualTypeArguments().length == 2) {
            Class<?> k = toClass(pt.getActualTypeArguments()[0]);
            Class<?> v = toClass(pt.getActualTypeArguments()[1]);
            if (k != null && v != null && k != Object.class && v != Object.class) {
                return CodeBlock.of("$T.entry($L, $L)", map, stubValue(k), stubValue(v));
            }
        }
        return CodeBlock.of("$T.entry(new Object(), new Object())", map);   // raw: Map.entry rejects null
    }

    /** A concrete {@code X.class} literal for a method returning {@code Class<...>} (Instancio can't
     *  build java.lang.Class), using the wildcard/type upper bound; Object.class when raw/unbounded. */
    private static CodeBlock classReturnStub(java.lang.reflect.Type gret) {
        if (toClass(gret) != Class.class) return null;
        Class<?> bound = Object.class;
        if (gret instanceof java.lang.reflect.ParameterizedType pt) {
            java.lang.reflect.Type arg = pt.getActualTypeArguments()[0];
            if (arg instanceof java.lang.reflect.WildcardType wt && wt.getUpperBounds().length > 0) {
                bound = toClass(wt.getUpperBounds()[0]);
            } else {
                bound = toClass(arg);
            }
        }
        if (bound == null) bound = Object.class;
        return CodeBlock.of("$T.class", ClassName.get(bound));
    }

    /** A real Optional stub value (present, with the element type) for a method returning Optional/
     *  OptionalInt/Long/Double, else null. A mocked Optional returns null from get()/map() -> caller
     *  NPEs; Optional.empty() throws on get(). A present Optional exercises get()/map()/orElse alike. */
    private static CodeBlock optionalReturnStub(java.lang.reflect.Type gret) {
        Class<?> raw = toClass(gret);
        if (raw == java.util.OptionalInt.class) return CodeBlock.of("$T.of(0)", raw);
        if (raw == java.util.OptionalLong.class) return CodeBlock.of("$T.of(0L)", raw);
        if (raw == java.util.OptionalDouble.class) return CodeBlock.of("$T.of(0.0)", raw);
        if (raw != java.util.Optional.class) return null;
        if (gret instanceof java.lang.reflect.ParameterizedType pt) {
            Class<?> elem = toClass(pt.getActualTypeArguments()[0]);
            if (elem != null && elem != Object.class) {
                return CodeBlock.of("$T.of($L)", java.util.Optional.class, stubValue(elem));
            }
        }
        return CodeBlock.of("$T.empty()", java.util.Optional.class);   // element unknown -> empty (still real)
    }

    /** Safe stub value for a non-void, non-primitive return: mock interfaces/abstract types
     *  (Instancio cannot instantiate them, so they would throw at runtime), Instancio.create concretes. */
    private static CodeBlock stubValue(Class<?> ret) {
        Class<?> sealedImpl = sealedConcrete(ret);
        if (sealedImpl != null) {   // sealed return -> permitted concrete impl (Mockito can't mock sealed)
            return CodeBlock.of("$T.create($T.class)", INSTANCIO, ClassName.get(sealedImpl));
        }
        if (ret.isArray()) {
            Class<?> comp = ret.getComponentType();
            if (comp.isInterface() || (!comp.isPrimitive() && !comp.isArray()
                    && java.lang.reflect.Modifier.isAbstract(comp.getModifiers()))) {
                return CodeBlock.of("new $T[0]", ClassName.get(comp));   // Instancio fills iface/abstract elems with null → NPE
            }
        }
        TypeName tn = javaPoetType(ret);
        Class<?> coll = collectionImpl(ret);
        if (coll != null) {   // empty real collection: a mocked List returns isEmpty()==false & get(i)==null → NPE
            return CodeBlock.of("new $T<>()", ClassName.get(coll));
        }
        // mock interfaces/abstracts AND generic types (Instancio can't build Foo<T>/Optional<T> raw).
        // arrays report isAbstract()==true but are not mockable — Instancio.create them.
        boolean mockable = !ret.isArray() && ret != Class.class   // Class is generic but Mockito refuses to mock it
                && (ret.isInterface() || java.lang.reflect.Modifier.isAbstract(ret.getModifiers())
                        || ret.getTypeParameters().length > 0);
        if (mockable) {
            return CodeBlock.of("$T.mock($T.class)", MOCKITO, tn);
        }
        return CodeBlock.of("$T.create($T.class)", INSTANCIO, tn);
    }

    private static CodeBlock anyMatchers(Class<?>[] params) {
        if (params.length == 0) {
            return CodeBlock.of("");
        }
        CodeBlock.Builder sb = CodeBlock.builder();
        for (int i = 0; i < params.length; i++) {
            if (i > 0) {
                sb.add(", ");
            }
            sb.add(typedAnyMatcher(params[i]));
        }
        return sb.build();
    }

    /** Type-aware matcher: primitives use anyX(); references use any(Type.class) so overloads disambiguate. */
    private static CodeBlock typedAnyMatcher(Class<?> p) {
        if (p == int.class)     return CodeBlock.of("$T.anyInt()", MATCHERS);
        if (p == long.class)    return CodeBlock.of("$T.anyLong()", MATCHERS);
        if (p == double.class)  return CodeBlock.of("$T.anyDouble()", MATCHERS);
        if (p == float.class)   return CodeBlock.of("$T.anyFloat()", MATCHERS);
        if (p == short.class)   return CodeBlock.of("$T.anyShort()", MATCHERS);
        if (p == byte.class)    return CodeBlock.of("$T.anyByte()", MATCHERS);
        if (p == char.class)    return CodeBlock.of("$T.anyChar()", MATCHERS);
        if (p == boolean.class) return CodeBlock.of("$T.anyBoolean()", MATCHERS);
        return CodeBlock.of("$T.any($T.class)", MATCHERS, javaPoetType(p));
    }

    private static String primitiveDefaultExpr(Class<?> prim) {
        if (prim == boolean.class) return "false";
        if (prim == char.class) return "'\\0'";
        if (prim == long.class) return "0L";
        if (prim == float.class) return "0f";
        if (prim == double.class) return "0.0";
        if (prim == byte.class) return "(byte) 0";     // doReturn(int 0) on byte-returning mock -> WrongTypeOfReturnValue
        if (prim == short.class) return "(short) 0";   // same for short
        return "0";   // int
    }

    private static List<Method> contractInstanceMethods(Class<?> sut) {
        List<Method> declared = filterInstance(sut.getDeclaredMethods());
        if (!isThin(declared)) {
            return dedupe(declared);
        }
        LinkedHashSet<Method> contract = new LinkedHashSet<>(declared);
        for (Class<?> iface : allInterfaces(sut)) {
            contract.addAll(filterInstance(iface.getMethods()));
        }
        Class<?> parent = sut.getSuperclass();
        if (parent != null && parent != Object.class) {
            contract.addAll(filterInstance(parent.getDeclaredMethods()));
        }
        return dedupe(new ArrayList<>(contract));
    }

    private static List<Method> contractStaticMethods(Class<?> sut) {
        List<Method> declared = filterStatic(sut.getDeclaredMethods());
        if (!declared.isEmpty()) {
            return dedupe(declared);
        }
        LinkedHashSet<Method> contract = new LinkedHashSet<>();
        for (Class<?> iface : allInterfaces(sut)) {
            contract.addAll(filterStatic(iface.getMethods()));
        }
        Class<?> parent = sut.getSuperclass();
        if (parent != null && parent != Object.class) {
            contract.addAll(filterStatic(parent.getDeclaredMethods()));
        }
        return dedupe(new ArrayList<>(contract));
    }

    private static boolean isThin(List<Method> declared) {
        if (declared.isEmpty()) {
            return true;
        }
        for (Method m : declared) {
            if (!looksLikeAccessor(m)) {
                return false;
            }
        }
        return true;
    }

    private static boolean looksLikeAccessor(Method m) {
        if (m.getParameterCount() != 0 || m.getReturnType() == void.class) {
            return false;
        }
        String n = m.getName();
        return n.startsWith("get") || n.startsWith("is");
    }

    private static List<Class<?>> allInterfaces(Class<?> type) {
        LinkedHashSet<Class<?>> out = new LinkedHashSet<>();
        Class<?> cur = type;
        while (cur != null && cur != Object.class) {
            out.addAll(Arrays.asList(cur.getInterfaces()));
            for (Class<?> iface : cur.getInterfaces()) {
                collectInterfaces(iface, out);
            }
            cur = cur.getSuperclass();
        }
        return new ArrayList<>(out);
    }

    private static void collectInterfaces(Class<?> iface, Set<Class<?>> out) {
        if (!out.add(iface)) {
            return;
        }
        for (Class<?> parent : iface.getInterfaces()) {
            collectInterfaces(parent, out);
        }
    }

    private static List<Method> filterInstance(Method[] methods) {
        List<Method> out = new ArrayList<>();
        for (Method m : methods) {
            if (include(m) && !java.lang.reflect.Modifier.isStatic(m.getModifiers())) {
                out.add(m);
            }
        }
        return out;
    }

    private static List<Method> filterStatic(Method[] methods) {
        List<Method> out = new ArrayList<>();
        for (Method m : methods) {
            if (include(m) && java.lang.reflect.Modifier.isStatic(m.getModifiers())) {
                out.add(m);
            }
        }
        return out;
    }

    private static List<Method> dedupe(List<Method> methods) {
        List<Method> out = new ArrayList<>();
        Set<String> seen = new HashSet<>();
        for (Method m : methods) {
            String key = m.getName() + Arrays.toString(m.getParameterTypes());
            if (seen.add(key)) {
                out.add(m);
            }
        }
        return out;
    }

    private static boolean include(Method m) {
        int mod = m.getModifiers();
        if (!java.lang.reflect.Modifier.isPublic(mod)) {
            return false;
        }
        if (m.isSynthetic() || m.isBridge()) {
            return false;
        }
        String n = m.getName();
        if (n.equals("equals") || n.equals("hashCode") || n.equals("toString") || n.equals("getClass")
                || n.equals("notify") || n.equals("notifyAll") || n.equals("wait")) {
            return false;
        }
        Class<?> dc = m.getDeclaringClass();
        if (dc == Enum.class
                || (dc.isEnum() && (n.equals("values") || n.equals("valueOf") || n.equals("compareTo")))) {
            return false;   // synthesized enum values()/valueOf/compareTo + java.lang.Enum internals
        }
        return true;
    }

    private static String safeName(Method m) {
        String base = m.getName();
        if (m.getParameterCount() == 0) {
            return base;
        }
        return base + "_" + m.getParameterCount() + "args";
    }

    private static String uniqueName(String base, Set<String> usedNames) {
        String name = base;
        int i = 2;
        while (!usedNames.add(name)) {
            name = base + "_" + i++;
        }
        return name;
    }

    private static String simpleParamName(Parameter p) {
        String n = p.getName();
        if (n == null || n.startsWith("arg")) {
            n = p.getType().getSimpleName();
        }
        return n.replaceAll("[^A-Za-z0-9_]", "");
    }

    private static Class<?> box(Class<?> primitive) {
        if (primitive == boolean.class) return Boolean.class;
        if (primitive == int.class) return Integer.class;
        if (primitive == long.class) return Long.class;
        if (primitive == double.class) return Double.class;
        if (primitive == float.class) return Float.class;
        if (primitive == short.class) return Short.class;
        if (primitive == byte.class) return Byte.class;
        if (primitive == char.class) return Character.class;
        return primitive;
    }

    // ---- static / construction isolation: mockStatic / mockConstruction in try-with-resources ----

    /** Parse the SUT method body for static-call targets and new'd types (project types only). */
    private static SutStatics analyzeStaticsAndNew(Class<?> sut, Method method) {
        try {
            Path src = Config.mainSourceRoot().resolve(sut.getName().replace('.', '/') + ".java");
            if (!Files.isRegularFile(src)) {
                return null;
            }
            CompilationUnit cu = StaticJavaParser.parse(src);
            String pkg = sut.getPackageName();
            MethodDeclaration target = null;
            for (MethodDeclaration md : cu.findAll(MethodDeclaration.class)) {
                if (md.getNameAsString().equals(method.getName())
                        && md.getParameters().size() == method.getParameterCount()) {
                    target = md;
                    break;
                }
            }
            if (target == null) {
                return null;
            }
            LinkedHashSet<Class<?>> statics = new LinkedHashSet<>();
            LinkedHashSet<Class<?>> constructed = new LinkedHashSet<>();
            for (MethodCallExpr call : target.findAll(MethodCallExpr.class)) {
                if (call.getScope().isPresent()) {
                    com.github.javaparser.ast.expr.Expression s = call.getScope().get();
                    String q = null;
                    if (s instanceof NameExpr ne && !ne.getNameAsString().isEmpty()
                            && Character.isUpperCase(ne.getNameAsString().charAt(0))) {
                        q = ne.getNameAsString();                       // simple: Class.method()
                    } else if (s instanceof com.github.javaparser.ast.expr.FieldAccessExpr fae) {
                        q = fae.toString();                             // fully-qualified: com.pkg.Class.method()
                    }
                    if (q != null) {
                        Class<?> c = q.contains(".") ? tryLoad(q) : resolveSimple(cu, pkg, q);
                        if (c != null && isProjectType(c) && !c.equals(sut)) {
                            statics.add(c);
                        }
                    }
                } else {
                    String mname = call.getNameAsString();              // no scope: static import?
                    for (ImportDeclaration imp : cu.getImports()) {
                        if (imp.isStatic() && !imp.isAsterisk() && imp.getNameAsString().endsWith("." + mname)) {
                            String cls = imp.getNameAsString().substring(0, imp.getNameAsString().lastIndexOf('.'));
                            Class<?> c = tryLoad(cls);
                            if (c != null && isProjectType(c) && !c.equals(sut)) {
                                statics.add(c);
                            }
                        }
                    }
                }
            }
            for (ObjectCreationExpr oce : target.findAll(ObjectCreationExpr.class)) {
                Class<?> c = resolveSimple(cu, pkg, oce.getType().getNameAsString());
                if (c != null && isProjectType(c) && !c.equals(sut)) {   // never mock-construct the SUT's own type
                    constructed.add(c);
                }
            }
            return new SutStatics(statics, constructed);
        } catch (Throwable t) {
            return null;
        }
    }

    private static Class<?> resolveSimple(CompilationUnit cu, String sutPkg, String simple) {
        for (ImportDeclaration imp : cu.getImports()) {
            if (!imp.isAsterisk() && imp.getNameAsString().endsWith("." + simple)) {
                Class<?> c = tryLoad(imp.getNameAsString());
                if (c != null) {
                    return c;
                }
            }
        }
        Class<?> c = tryLoad(sutPkg + "." + simple);
        return (c != null) ? c : tryLoad("java.lang." + simple);
    }

    private static Class<?> tryLoad(String fqn) {
        try {
            return Class.forName(fqn, false, TestSkeletonGenerator.class.getClassLoader());
        } catch (Throwable t) {
            return null;
        }
    }

    /** try-with-resources header: one MockedStatic per static target, one MockedConstruction per new'd type. */
    private static CodeBlock mockScopeResources(SutStatics st) {
        CodeBlock.Builder r = CodeBlock.builder();
        boolean first = true;
        for (Class<?> x : st.staticTargets()) {
            if (!first) {
                r.add(";\n");
            }
            first = false;
            r.add("$T<$T> $L = $T.mockStatic($T.class)",
                    MOCKED_STATIC, ClassName.get(x), msVar(x), MOCKITO, ClassName.get(x));
        }
        for (Class<?> y : st.constructedTypes()) {
            if (!first) {
                r.add(";\n");
            }
            first = false;
            CodeBlock init = minimalStubs(new MockRef(y, "cm"));   // stub the constructed mock's own methods
            if (init.isEmpty()) {
                r.add("$T<$T> $L = $T.mockConstruction($T.class)",
                        MOCKED_CONSTRUCTION, ClassName.get(y), mcVar(y), MOCKITO, ClassName.get(y));
            } else {   // 2-arg form: else new T() returns a mock whose methods return null -> caller NPEs
                r.add("$T<$T> $L = $T.mockConstruction($T.class, (cm, context) -> {\n$>$L$<})",
                        MOCKED_CONSTRUCTION, ClassName.get(y), mcVar(y), MOCKITO, ClassName.get(y), init);
            }
        }
        return r.build();
    }

    /** Stub each static target's non-void static methods so the isolated call graph does not NPE. */
    private static void emitStaticStubs(MethodSpec.Builder b, SutStatics st, Class<?> sut) {
        for (Class<?> x : st.staticTargets()) {
            for (Method sm : x.getDeclaredMethods()) {
                if (!java.lang.reflect.Modifier.isStatic(sm.getModifiers()) || !include(sm)) {
                    continue;
                }
                java.lang.reflect.Type gret = sm.getGenericReturnType();
                Class<?> ret = sm.getReturnType();
                if (ret == void.class) {
                    continue;
                }
                CodeBlock matchers = anyMatchers(sm.getParameterTypes());
                if (ret.isPrimitive()) {
                    b.addStatement("$L.when(() -> $T.$L($L)).thenReturn($L)",
                            msVar(x), ClassName.get(x), sm.getName(), matchers, primitiveDefaultExpr(ret));
                } else if (isTypeVarReturn(gret)) {   // unresolved static type-var: null (a bare Object would CCE)
                    b.addStatement("$L.when(() -> $T.$L($L)).thenReturn(null)",
                            msVar(x), ClassName.get(x), sm.getName(), matchers);
                } else {
                    CodeBlock rv = classReturnStub(gret);   // Class<...> -> X.class
                    if (rv == null) rv = optionalReturnStub(gret);   // real Optional, not mock
                    if (rv == null) rv = entryReturnStub(gret);   // real Map.entry, not mock
                    if (rv == null) rv = futureReturnStub(gret);   // real completed future, not mock
                    if (rv == null) rv = stubValue(ret);
                    b.addStatement("$L.when(() -> $T.$L($L)).thenReturn($L)",
                            msVar(x), ClassName.get(x), sm.getName(), matchers, rv);
                }
            }
            b.addCode(staticFactoryStubs(sut, x));   // call-site X.class binds a static <T> T m(Class<T>) factory
        }
    }

    /** Static analog of {@link #factoryStubs}: for {@code X.m(Y.class)} call sites on a static factory,
     *  emit {@code msVar.when(() -> X.m(eq(Y.class))).thenReturn(<value Y>)}. */
    private static CodeBlock staticFactoryStubs(Class<?> sut, Class<?> collab) {
        CodeBlock.Builder out = CodeBlock.builder();
        try {
            Path src = Config.mainSourceRoot().resolve(sut.getName().replace('.', '/') + ".java");
            if (!Files.isRegularFile(src)) return out.build();
            CompilationUnit cu = StaticJavaParser.parse(src);
            String pkg = sut.getPackageName();
            java.util.Set<String> seen = new java.util.HashSet<>();
            for (MethodCallExpr call : cu.findAll(MethodCallExpr.class)) {
                if (call.getScope().isEmpty()) continue;
                com.github.javaparser.ast.expr.Expression scope = call.getScope().get();
                String scopeName = scope instanceof NameExpr ne ? ne.getNameAsString() : scope.toString();
                if (!collab.getSimpleName().equals(scopeName) && !collab.getName().equals(scopeName)) continue;
                int[] pos = {-1};
                Class<?> y = classArg(call, cu, pkg, pos);
                if (y == null) continue;
                Method fm = factoryMethod(collab, call.getNameAsString(), call.getArguments().size(), pos[0]);
                if (fm == null || !java.lang.reflect.Modifier.isStatic(fm.getModifiers())) continue;
                if (!seen.add(call.getNameAsString() + "|" + pos[0] + "|" + y.getName())) continue;
                out.addStatement("$L.when(() -> $T.$L($L)).thenReturn($L)",
                        msVar(collab), ClassName.get(collab), call.getNameAsString(),
                        factoryMatchers(fm, pos[0], y), stubValue(y));
            }
        } catch (Throwable ignored) {
            // best-effort
        }
        return out.build();
    }

    private static String msVar(Class<?> x) {
        return "ms" + x.getSimpleName();
    }

    private static String mcVar(Class<?> y) {
        return "mc" + y.getSimpleName();
    }

    /**
     * TG-BOILERDET: detected facades never get mockStatic/mockConstruction — the delegate-swap
     * fixture controls them, and a static mock of the facade is exactly the collision the play
     * exists to remove.
     */
    private static SutStatics suppressFacades(SutStatics st, Set<Class<?>> facadeSuppress) {
        if (st == null || facadeSuppress.isEmpty()) {
            return st;
        }
        LinkedHashSet<Class<?>> statics = new LinkedHashSet<>(st.staticTargets());
        LinkedHashSet<Class<?>> constructed = new LinkedHashSet<>(st.constructedTypes());
        statics.removeAll(facadeSuppress);
        constructed.removeAll(facadeSuppress);
        return new SutStatics(statics, constructed);
    }

    private record SutStatics(Set<Class<?>> staticTargets, Set<Class<?>> constructedTypes) {}

    private record MockRef(Class<?> type, String varName) {}

    private record FieldMockRef(Class<?> type, String varName, Class<?> declaringClass, String fieldName,
                                boolean skipStub) {}
}

    private CallChainScanner() {}

    public static void run() throws Exception {
        Path outDir = Config.reportsDir().resolve("callchain");
        Files.createDirectories(outDir);
        Path root = Config.mainSourceRoot();
        int maxDepth = Math.max(1, Config.CALL_CHAIN_MAX_DEPTH);
        int wrote = 0;
        for (String target : Config.readMethodTargets()) {
            try {
                String fqcn = target;
                String method = null;
                int hash = target.indexOf('#');
                if (hash > 0) {
                    fqcn = target.substring(0, hash).trim();
                    method = target.substring(hash + 1).trim();
                }
                if (method == null || method.isBlank()) {
                    java.util.Map<String,Integer> seen = new java.util.HashMap<>();
                    for (String m : listPublicMethods(root, fqcn)) {
                        int n = seen.merge(m, 1, Integer::sum);
                        String fileKey = (n == 1) ? m : m + "_" + n;   // BUG-1: overloads must not overwrite
                        wrote += writeOne(outDir, root, fqcn, m, fileKey, maxDepth) ? 1 : 0;
                    }
                } else {
                    wrote += writeOne(outDir, root, fqcn, method, method, maxDepth) ? 1 : 0;
                }
            } catch (Throwable t) {   // isolate per-target: one bad target must not abort the batch
                System.out.println("SKIP callchain " + target + " -> " + t.getClass().getSimpleName()
                        + ": " + t.getMessage());
            }
        }
        System.out.println("CallChain done. " + wrote + " report(s) in " + outDir.toAbsolutePath());
    }

    private static boolean writeOne(Path outDir, Path root, String fqcn, String method, String fileKey, int maxDepth)
            throws Exception {
        String md = analyze(root, fqcn, method, maxDepth);
        Path out = outDir.resolve(fqcn + "__" + fileKey + ".md");   // BUG-1: fileKey disambiguates overloads
        Files.writeString(out, md);
        System.out.println("Wrote " + out);
        return true;
    }

    public static String analyze(Path mainSourceRoot, String fqcn, String method, int maxDepth) throws Exception {
        StringBuilder sb = new StringBuilder();
        sb.append("# callchain\n");
        sb.append("sut: ").append(fqcn).append('#').append(method).append('\n');
        sb.append("maxDepth: ").append(maxDepth).append('\n');
        sb.append('\n');
        List<String> lines = new ArrayList<>();
        walk(mainSourceRoot, fqcn, method, 0, maxDepth, new LinkedHashSet<>(), lines, "");
        if (lines.isEmpty()) {
            sb.append("- (no outbound calls found or source missing)\n");
        } else {
            for (String line : lines) {
                sb.append(line).append('\n');
            }
        }
        sb.append('\n');
        sb.append("## notes\n");
        sb.append("- Static only; no reflection/DI/poly resolution.\n");
        return sb.toString();
    }

    private static void walk(
            Path root,
            String fqcn,
            String method,
            int depth,
            int maxDepth,
            Set<String> stack,
            List<String> out,
            String indent) throws Exception {
        if (depth >= maxDepth) {
            return;
        }
        String key = fqcn + "#" + method;
        if (!stack.add(key)) {
            out.add(indent + "- (cycle " + key + ")");
            return;
        }
        Path src = sourcePath(root, fqcn);
        if (!Files.isRegularFile(src)) {
            out.add(indent + "- (no source " + fqcn + ")");
            stack.remove(key);
            return;
        }
        CompilationUnit cu = StaticJavaParser.parse(src);
        Optional<MethodDeclaration> md = findMethod(cu, simpleName(fqcn), method);
        if (md.isEmpty()) {
            out.add(indent + "- (method not found " + key + ")");
            stack.remove(key);
            return;
        }
        String pkg = cu.getPackageDeclaration().map(p -> p.getNameAsString()).orElse("");
        Map<String, String> fieldTypes = fieldTypes(cu, simpleName(fqcn));
        Map<String, String> localTypes = localVarTypes(md.get(), cu, pkg);
        for (ObjectCreationExpr oce : md.get().findAll(ObjectCreationExpr.class)) {
            String t = resolveTypeName(cu, pkg, oce.getType().getNameAsString());
            String loc = src.getFileName() + ":" + oce.getBegin().map(p -> p.line).orElse(-1);
            out.add(indent + "- " + simpleName(fqcn) + "." + method + " -> new " + simpleName(t) + " (" + loc + ")");
        }
        int n = 0;
        for (MethodCallExpr call : md.get().findAll(MethodCallExpr.class)) {
            if (n++ >= Config.CALL_CHAIN_MAX_EDGES) {
                out.add(indent + "- … capped");
                break;
            }
            String calleeName = call.getNameAsString();
            String scopeType = resolveScopeType(call, root, fieldTypes, localTypes, cu, pkg, fqcn);
            String loc = src.getFileName() + ":" + call.getBegin().map(p -> p.line).orElse(-1);
            String edge = simpleName(fqcn) + "." + method + " -> "
                    + (scopeType != null ? simpleName(scopeType) + "." : "")
                    + calleeName + " (" + loc + ")";
            out.add(indent + "- " + edge);
            if (scopeType != null && !scopeType.startsWith("java.") && depth + 1 < maxDepth) {
                walk(root, scopeType, calleeName, depth + 1, maxDepth, stack, out, indent + "  ");
            }
        }
        stack.remove(key);
    }

    private static String resolveScopeType(MethodCallExpr call, Path root, Map<String, String> fieldTypes,
            Map<String, String> localTypes, CompilationUnit cu, String pkg, String ownerFqcn) {
        Optional<Expression> scope = call.getScope();
        if (scope.isEmpty()) {
            return ownerFqcn; // same class / this
        }
        Expression s = scope.get();
        if (s instanceof NameExpr ne) {
            String n = ne.getNameAsString();
            if ("this".equals(n)) {
                return ownerFqcn;
            }
            if (localTypes.containsKey(n)) {
                return localTypes.get(n);   // local variable / parameter receiver
            }
            if (fieldTypes.containsKey(n)) {
                return fieldTypes.get(n);
            }
            if (!n.isEmpty() && Character.isUpperCase(n.charAt(0))) {
                return resolveTypeName(cu, pkg, n);   // static-class receiver
            }
            return null;
        }
        if (s instanceof FieldAccessExpr fa && "this".equals(fa.getScope().toString())) {
            return fieldTypes.get(fa.getNameAsString());
        }
        if (s instanceof MethodCallExpr inner) {
            // chained call a().b(): resolve inner receiver, then look up inner's DECLARED return type
            String innerOwner = resolveScopeType(inner, root, fieldTypes, localTypes, cu, pkg, ownerFqcn);
            if (innerOwner != null) {
                return returnTypeOf(root, innerOwner, inner.getNameAsString());
            }
        }
        return null;
    }

    /** Declared return type of ownerFqcn#methodName, resolved to a FQN (deterministic, from source). */
    private static String returnTypeOf(Path root, String ownerFqcn, String methodName) {
        try {
            Path src = sourcePath(root, ownerFqcn);
            if (!Files.isRegularFile(src)) {
                return null;
            }
            CompilationUnit ocu = StaticJavaParser.parse(src);
            String opkg = ocu.getPackageDeclaration().map(p -> p.getNameAsString()).orElse("");
            for (MethodDeclaration m : ocu.findAll(MethodDeclaration.class)) {
                if (m.getNameAsString().equals(methodName)) {
                    String rt = m.getType().asString();
                    return "void".equals(rt) ? null : resolveTypeName(ocu, opkg, rt);
                }
            }
        } catch (Throwable ignored) { }
        return null;
    }

    /** Local variable + parameter names -> resolved (erased) type, for receiver resolution. */
    private static Map<String, String> localVarTypes(MethodDeclaration md, CompilationUnit cu, String pkg) {
        Map<String, String> map = new LinkedHashMap<>();
        for (VariableDeclarationExpr v : md.findAll(VariableDeclarationExpr.class)) {
            v.getVariables().forEach(var -> {
                String t = var.getType().asString().replaceAll("<.*>", "").trim();
                if (!t.isEmpty() && Character.isJavaIdentifierStart(t.charAt(0))) {
                    map.put(var.getNameAsString(), resolveTypeName(cu, pkg, t));
                }
            });
        }
        md.getParameters().forEach(p -> {
            String t = p.getType().asString().replaceAll("<.*>", "").trim();
            if (!t.isEmpty() && Character.isJavaIdentifierStart(t.charAt(0))) {
                map.put(p.getNameAsString(), resolveTypeName(cu, pkg, t));
            }
        });
        return map;
    }

    /** Resolve a simple type name to a FQN via imports, then java.lang, else same package. */
    private static String resolveTypeName(CompilationUnit cu, String pkg, String simple) {
        String bare = simple.replaceAll("<.*>", "").replace("[]", "").trim();
        if (bare.contains(".")) {
            return bare;
        }
        for (ImportDeclaration imp : cu.getImports()) {
            if (!imp.isAsterisk() && imp.getNameAsString().endsWith("." + bare)) {
                return imp.getNameAsString();
            }
        }
        if (isJavaLang(bare)) {
            return "java.lang." + bare;
        }
        return (bare.isEmpty() || pkg.isEmpty()) ? bare : pkg + "." + bare;
    }

    private static boolean isJavaLang(String simple) {
        try {
            Class.forName("java.lang." + simple);
            return true;
        } catch (Throwable t) {
            return false;
        }
    }

    private static Map<String, String> fieldTypes(CompilationUnit cu, String simple) {
        Map<String, String> map = new LinkedHashMap<>();
        for (ClassOrInterfaceDeclaration c : cu.findAll(ClassOrInterfaceDeclaration.class)) {
            if (!c.getNameAsString().equals(simple)) {
                continue;
            }
            c.getFields().forEach(f -> f.getVariables().forEach(v -> {
                String type = f.getElementType().asString().replaceAll("<.*>", "").trim();  // erase generics
                String pkg = cu.getPackageDeclaration().map(p -> p.getNameAsString()).orElse("");
                if (type.isEmpty() || !Character.isJavaIdentifierStart(type.charAt(0))) {
                    return;
                }
                // Route through the import-aware resolver like every other path; the old inline
                // pkg-concat mis-packaged imported field types (TG-HOP2NAMES root cause).
                map.put(v.getNameAsString(), resolveTypeName(cu, pkg, type));
            }));
        }
        return map;
    }

    private static Optional<MethodDeclaration> findMethod(CompilationUnit cu, String simple, String method) {
        for (ClassOrInterfaceDeclaration c : cu.findAll(ClassOrInterfaceDeclaration.class)) {
            if (!c.getNameAsString().equals(simple)) {
                continue;
            }
            for (MethodDeclaration m : c.getMethods()) {
                if (m.getNameAsString().equals(method)) {
                    return Optional.of(m);
                }
            }
        }
        return Optional.empty();
    }

    static List<String> listPublicMethods(Path root, String fqcn) throws Exception {
        Path src = sourcePath(root, fqcn);
        List<String> names = new ArrayList<>();
        if (!Files.isRegularFile(src)) {
            return names;
        }
        CompilationUnit cu = StaticJavaParser.parse(src);
        String simple = simpleName(fqcn);
        for (ClassOrInterfaceDeclaration c : cu.findAll(ClassOrInterfaceDeclaration.class)) {
            if (!c.getNameAsString().equals(simple)) {
                continue;
            }
            for (MethodDeclaration m : c.getMethods()) {
                if (m.isPublic() && !m.getNameAsString().equals("main")) {
                    names.add(m.getNameAsString());
                }
            }
        }
        return names;
    }

    private static Path sourcePath(Path root, String fqcn) {
        String clean = fqcn.replaceAll("<.*>", "").trim();   // never allow a generic/illegal name in a path
        Path direct = root.resolve(clean.replace('.', '/') + ".java");
        if (Files.isRegularFile(direct)) {
            return direct;
        }
        // Nested types live in their OUTER class's file (TG-HOP2NAMES: a.b.Outer.Inner or
        // Outer$Inner has no Inner.java). Walk trailing segments up until a file exists; the
        // declaration is then found by simple name inside that compilation unit.
        String walk = clean.replace('$', '.');
        int cut;
        while ((cut = walk.lastIndexOf('.')) > 0) {
            walk = walk.substring(0, cut);
            Path outer = root.resolve(walk.replace('.', '/') + ".java");
            if (Files.isRegularFile(outer)) {
                return outer;
            }
        }
        // Resolvers emit package+Simple for same-file nested types, so the outer name is not in
        // the FQCN at all. Last resort: scan the package directory for the file DECLARING the
        // simple name (bounded to one directory; deterministic first match in sorted order).
        int lastDot = clean.lastIndexOf('.');
        if (lastDot > 0) {
            Path pkgDir = root.resolve(clean.substring(0, lastDot).replace('.', '/'));
            String simple = clean.substring(lastDot + 1);
            if (Files.isDirectory(pkgDir)) {
                try (var files = Files.list(pkgDir)) {
                    for (Path candidate : files.filter(f -> f.toString().endsWith(".java"))
                            .sorted().toList()) {
                        try {
                            CompilationUnit cu = StaticJavaParser.parse(candidate);
                            if (cu.findAll(ClassOrInterfaceDeclaration.class).stream()
                                    .anyMatch(c -> c.getNameAsString().equals(simple))) {
                                return candidate;
                            }
                        } catch (Throwable ignored) { }
                    }
                } catch (Throwable ignored) { }
            }
        }
        return direct;   // original (missing) path: callers report "(no source ...)" honestly
    }

    private static String simpleName(String fqcn) {
        int i = fqcn.lastIndexOf('.');
        return i < 0 ? fqcn : fqcn.substring(i + 1);
    }

/**
 * TG-BOILERDET detection pass: statically decide, per SUT, whether the test needs the
 * delegate-swap boilerplate (mockito-static-lab LoggerFixture/ConfigFixture play).
 *
 * A FACADE is a project class F with a static accessor returning F (singleton shape) and at
 * least one swappable delegate field (non-final static, or instance, reference-typed state the
 * lab play swaps and restores). A SUT CONSUMES a facade when the SUT, a superclass, or a hop-1
 * collaborator (field / constructor-param project type) declares a field of type F or calls
 * F's static accessor in source — this catches config-in-constructor and static-init reads.
 *
 * Pure detection: no source is modified; the skeleton generator decides what to emit.
 *
 * WHY THESE TARGETS - the three-leg test. Most statics are harmless furniture; a fixture is
 * needed only when all three legs hold:
 *   1. SHARED    - static state: one copy per JVM, every test touches the same one.
 *   2. MUTABLE   - it can change after birth: a swappable delegate, a map, settable levels.
 *   3. CONSULTED - the behaviour under test actually reads or writes it.
 * Static methods hold no state (nothing leaks); static final constants cannot change (nothing
 * to undo); static nested classes are naming only. Logger, config, cache and JNDI registries
 * pass all three legs - that family is what this scanner hunts, and the three legs are exactly
 * the checks encoded in facadeShape/swappableDelegates/consumer detection below.
 */
public final class SMCFixtureScanner {

    private SMCFixtureScanner() {}

    /** One swappable delegate on a facade. */
    public record DelegateField(String name, Class<?> type, boolean staticField, boolean mapShaped) {}

    /** One detected facade with its accessor and swappable delegates. */
    public record Facade(Class<?> facadeClass, Method accessor, List<DelegateField> delegates) {

        /** True when any delegate is instance-scoped, so the fixture must fetch the singleton. */
        boolean needsInstance() {
            return delegates.stream().anyMatch(d -> !d.staticField());
        }
    }

    /**
     * Detects every facade the SUT-set consumes. Deterministic order: SUT first, then parents,
     * then collaborators in declaration order.
     */
    public static List<Facade> detect(Class<?> sut) {
        LinkedHashSet<Class<?>> consumerSet = consumerSet(sut);
        LinkedHashSet<Class<?>> facadeClasses = new LinkedHashSet<>();

        for (Class<?> consumer : consumerSet) {
            collectFacadesFromFields(consumer, facadeClasses);
            collectFacadesFromSource(consumer, facadeClasses);
            collectFacadesFromSignatures(consumer, facadeClasses);
        }
        facadeClasses.remove(sut);   // a facade never needs a fixture inside its own test

        List<Facade> out = new ArrayList<>();
        for (Class<?> f : facadeClasses) {
            Facade facade = asFacade(f);
            if (facade != null) {
                out.add(facade);
            }
        }
        return out;
    }

    /** Facade classes for one SUT as a set, for suppressing mockStatic on them. */
    public static Set<Class<?>> facadeClasses(List<Facade> facades) {
        LinkedHashSet<Class<?>> out = new LinkedHashSet<>();
        for (Facade f : facades) {
            out.add(f.facadeClass());
        }
        return out;
    }

    /** SUT + superclass chain (stop at java.*) + hop-1 collaborator project types. */
    private static LinkedHashSet<Class<?>> consumerSet(Class<?> sut) {
        LinkedHashSet<Class<?>> out = new LinkedHashSet<>();
        Class<?> cur = sut;
        while (cur != null && cur != Object.class && !cur.getName().startsWith("java.")) {
            out.add(cur);
            cur = cur.getSuperclass();
        }
        for (Class<?> base : new ArrayList<>(out)) {
            for (Field f : base.getDeclaredFields()) {
                if (!f.isSynthetic() && isProjectClass(baseType(f.getType()))) {
                    out.add(baseType(f.getType()));
                }
            }
            for (var ctor : base.getDeclaredConstructors()) {
                for (Class<?> p : ctor.getParameterTypes()) {
                    if (isProjectClass(baseType(p))) {
                        out.add(baseType(p));
                    }
                }
            }
        }
        return out;
    }

    /** Field of a facade-shaped project type anywhere in the consumer → facade hit. */
    private static void collectFacadesFromFields(Class<?> consumer, Set<Class<?>> out) {
        for (Field f : consumer.getDeclaredFields()) {
            if (f.isSynthetic()) {
                continue;
            }
            Class<?> ft = baseType(f.getType());
            if (ft != consumer && isProjectClass(ft) && asFacade(ft) != null) {
                out.add(ft);
            }
        }
    }

    /**
     * TWEAK-POINT (detection): static call {@code F.accessor(...)} in the consumer's source —
     * simple name (resolved through imports/package) OR fully-qualified — and facade-typed
     * LOCAL VARIABLE declarations. Method bodies are the reliable gun when no field exists.
     */
    private static void collectFacadesFromSource(Class<?> consumer, Set<Class<?>> out) {
        Optional<CompilationUnit> cu = parseSource(consumer);
        if (cu.isEmpty()) {
            return;
        }
        String pkg = consumer.getPackageName();
        for (MethodCallExpr call : cu.get().findAll(MethodCallExpr.class)) {
            if (call.getScope().isEmpty()) {
                continue;
            }
            Class<?> resolved = null;
            if (call.getScope().get() instanceof NameExpr ne) {
                String scope = ne.getNameAsString();
                if (!scope.isEmpty() && Character.isUpperCase(scope.charAt(0))) {
                    resolved = resolveSimple(cu.get(), pkg, scope);
                }
            } else if (call.getScope().get() instanceof com.github.javaparser.ast.expr.FieldAccessExpr fae) {
                resolved = tryLoad(fae.toString());   // fully-qualified accessor call, no import needed
            }
            if (resolved == null || resolved == consumer || !isProjectClass(resolved)) {
                continue;
            }
            Facade facade = asFacade(resolved);
            if (facade != null && facade.accessor().getName().equals(call.getNameAsString())) {
                out.add(resolved);
            }
        }
        for (com.github.javaparser.ast.body.VariableDeclarator var
                : cu.get().findAll(com.github.javaparser.ast.body.VariableDeclarator.class)) {
            String typeName = var.getType().asString().replaceAll("<.*>", "").trim();
            Class<?> resolved = typeName.contains(".") ? tryLoad(typeName)
                    : resolveSimple(cu.get(), pkg, typeName);
            if (resolved != null && resolved != consumer && isProjectClass(resolved)
                    && asFacade(resolved) != null) {
                out.add(resolved);
            }
        }
    }

    /**
     * TWEAK-POINT (detection): facade types arriving through SIGNATURES — constructor params,
     * method params, and return types (reflection; no source needed).
     */
    private static void collectFacadesFromSignatures(Class<?> consumer, Set<Class<?>> out) {
        java.util.List<Class<?>> candidates = new ArrayList<>();
        for (var ctor : consumer.getDeclaredConstructors()) {
            candidates.addAll(java.util.Arrays.asList(ctor.getParameterTypes()));
        }
        for (Method m : consumer.getDeclaredMethods()) {
            candidates.addAll(java.util.Arrays.asList(m.getParameterTypes()));
            candidates.add(m.getReturnType());
        }
        for (Class<?> c : candidates) {
            Class<?> base = baseType(c);
            if (base != consumer && isProjectClass(base) && asFacade(base) != null) {
                out.add(base);
            }
        }
    }


    /**
     * Facade shape test. Returns the facade model, or null when the class does not qualify or
     * its accessor takes parameters the fixture cannot synthesize (Class/String only).
     */
    private static Facade asFacade(Class<?> candidate) {
        if (candidate == null || !isProjectClass(candidate)) {
            return null;
        }
        Method accessor = singletonAccessor(candidate);
        if (accessor == null) {
            return null;
        }
        if (!hasSelfTypedHolder(candidate)) {
            return null;   // D4: factory shape (fresh object per call) - swapping it is a no-op
        }
        List<DelegateField> delegates = swappableDelegates(candidate);
        if (delegates.isEmpty()) {
            if (hasFrozenDelegates(candidate)) {
                System.out.println("NOTE facade-like but frozen (static-final delegates only), "
                        + "unsupported: " + candidate.getName());
            }
            return null;
        }
        return new Facade(candidate, accessor, delegates);
    }

    /** Public static method returning the class itself, with only synthesizable params. */
    private static Method singletonAccessor(Class<?> candidate) {
        for (Method m : candidate.getDeclaredMethods()) {
            if (!Modifier.isStatic(m.getModifiers()) || !Modifier.isPublic(m.getModifiers())
                    || m.isSynthetic() || m.getReturnType() != candidate) {
                continue;
            }
            boolean synthesizable = true;
            for (Class<?> p : m.getParameterTypes()) {
                if (p != Class.class && p != String.class) {
                    synthesizable = false;
                }
            }
            if (synthesizable) {
                return m;
            }
        }
        return null;
    }

    /**
     * Swappable delegates: reference-typed mutable state the play controls. Static-final is
     * frozen (unswappable); value-shaped types (primitive/boxed/String/enum) are config data,
     * not delegates; the self-typed INSTANCE field is the singleton itself.
     */
    private static List<DelegateField> swappableDelegates(Class<?> candidate) {
        List<DelegateField> out = new ArrayList<>();
        for (Field f : candidate.getDeclaredFields()) {
            if (f.isSynthetic()) {
                continue;
            }
            boolean statik = Modifier.isStatic(f.getModifiers());
            if (statik && Modifier.isFinal(f.getModifiers())) {
                continue;
            }
            Class<?> ft = f.getType();
            if (ft == candidate || ft.isPrimitive() || ft.isEnum() || isValueShaped(ft)) {
                continue;
            }
            if (ft == Class.class) {
                continue;   // D3: Mockito cannot mock java.lang.Class - not a swappable delegate
            }
            out.add(new DelegateField(f.getName(), ft, statik, java.util.Map.class.isAssignableFrom(ft)));
        }
        return out;
    }


    /** D4 guard: a real singleton holds itself in a static field; factories do not. */
    private static boolean hasSelfTypedHolder(Class<?> candidate) {
        for (Field f : candidate.getDeclaredFields()) {
            if (Modifier.isStatic(f.getModifiers()) && f.getType() == candidate) {
                return true;
            }
        }
        return false;
    }

    /** D5 visibility: reference-typed static-final state exists but is frozen by contract. */
    private static boolean hasFrozenDelegates(Class<?> candidate) {
        for (Field f : candidate.getDeclaredFields()) {
            if (Modifier.isStatic(f.getModifiers()) && Modifier.isFinal(f.getModifiers())
                    && !f.getType().isPrimitive() && f.getType() != candidate
                    && !f.isSynthetic()) {
                return true;
            }
        }
        return false;
    }

    private static boolean isValueShaped(Class<?> type) {
        return type == String.class || type == Integer.class || type == Long.class
                || type == Double.class || type == Float.class || type == Short.class
                || type == Byte.class || type == Boolean.class || type == Character.class;
    }

    /** Project class iff its .java source exists under the main source root. */
    private static boolean isProjectClass(Class<?> type) {
        if (type == null || type.isPrimitive()
                || type.getPackageName().startsWith("java.")
                || type.getPackageName().startsWith("javax.")) {
            return false;
        }
        return Files.isRegularFile(sourcePath(type));
    }

    private static Path sourcePath(Class<?> type) {
        Class<?> outer = type;
        while (outer.getEnclosingClass() != null) {
            outer = outer.getEnclosingClass();
        }
        return Config.mainSourceRoot()
                .resolve(outer.getPackageName().replace('.', '/'))
                .resolve(outer.getSimpleName() + ".java");
    }

    private static Optional<CompilationUnit> parseSource(Class<?> type) {
        Path src = sourcePath(type);
        if (!Files.isRegularFile(src)) {
            return Optional.empty();
        }
        try {
            return Optional.of(StaticJavaParser.parse(src));
        } catch (Exception e) {
            return Optional.empty();
        }
    }

    private static Class<?> resolveSimple(CompilationUnit cu, String pkg, String simple) {
        for (ImportDeclaration imp : cu.getImports()) {
            if (!imp.isAsterisk() && !imp.isStatic() && imp.getNameAsString().endsWith("." + simple)) {
                Class<?> c = tryLoad(imp.getNameAsString());
                if (c != null) {
                    return c;
                }
            }
        }
        return tryLoad(pkg + "." + simple);
    }

    private static Class<?> tryLoad(String fqcn) {
        try {
            return Class.forName(fqcn, false, SMCFixtureScanner.class.getClassLoader());
        } catch (Throwable t) {
            return null;
        }
    }

    private static Class<?> baseType(Class<?> type) {
        Class<?> cur = type;
        while (cur != null && cur.isArray()) {
            cur = cur.getComponentType();
        }
        return cur;
    }
}

 * shaped like the lab's LoggerFixture/ConfigFixture (delegate swap in install, mandatory restore
 * in tearDown). Reflection stays INSIDE the fixture so emitted test code carries none, and no

 *

public final class FixtureGenerator {

    private static final ClassName MOCKITO = ClassName.get("org.mockito", "Mockito");

    private static final ClassName LINKED_MAP = ClassName.get("java.util", "LinkedHashMap");

    private FixtureGenerator() {}

    public static String fixtureSimpleName(SMCFixtureScanner.Facade facade) {
        return facade.facadeClass().getSimpleName() + "Fixture";
    }

    /** Writes the fixture unless one already exists; returns its ClassName either way. */

        ClassName fixtureName = ClassName.get(FIXTURE_PACKAGE, simpleName);
        Path target = testSourceRoot.resolve(FIXTURE_PACKAGE).resolve(simpleName + ".java");
        if (Files.exists(target)) {
            System.out.println("SKIP fixture (exists): " + target);
            return fixtureName;
        }

        JavaFile.builder(FIXTURE_PACKAGE,
                        buildType(facade, simpleName).addModifiers(Modifier.PUBLIC).build())
                .skipJavaLangImports(true)
                .addFileComment("AUTO-GENERATED by testgen FixtureGenerator (TG-BOILERDET)")
                .build()
                .writeTo(testSourceRoot);
        System.out.println("Wrote " + FIXTURE_PACKAGE + "." + simpleName);
        return fixtureName;
    }

    /**
     * Inline mode (BOILERPLATE_INLINE): the SAME fixture as a private static nested class of the
     * test class - self-contained file, no shared testsupport code.
     */
    public static TypeSpec nestedType(SMCFixtureScanner.Facade facade) {
        return buildType(facade, fixtureSimpleName(facade))
                .addModifiers(Modifier.PRIVATE, Modifier.STATIC)
                .build();
    }

    private static TypeSpec.Builder buildType(SMCFixtureScanner.Facade facade, String simpleName) {
        TypeSpec.Builder fixture = TypeSpec.classBuilder(simpleName)
                .addModifiers(Modifier.FINAL)   // visibility added by each emission mode
                .addJavadoc("AUTO-GENERATED delegate-swap fixture for $L.\n"
                                + "Lab play: swap the singleton's delegates, restore in tearDown — no mockStatic.\n"
                                + "Reflection is contained here so the test class carries none.\n",
                        facade.facadeClass().getName());

        fixture.addField(FieldSpec.builder(
                        ParameterizedTypeName.get(MAP, ClassName.get(String.class), ClassName.get(Object.class)),
                        "originals", Modifier.PRIVATE, Modifier.FINAL)
                .initializer("new $T<>()", LINKED_MAP)
                .build());

        for (SMCFixtureScanner.DelegateField d : facade.delegates()) {
            fixture.addField(FieldSpec.builder(handleType(d), handleName(d), Modifier.PRIVATE).build());
            fixture.addMethod(MethodSpec.methodBuilder(handleName(d))
                    .addModifiers(Modifier.PUBLIC)
                    .returns(handleType(d))
                    .addStatement("return $L", handleName(d))
                    .build());
        }

        fixture.addMethod(installMethod(facade));
        fixture.addMethod(restoreMethod(facade));
        return fixture;
    }

    /** install(): capture each original delegate, swap in the test double. */
    private static MethodSpec installMethod(SMCFixtureScanner.Facade facade) {
        MethodSpec.Builder b = MethodSpec.methodBuilder("install")
                .addModifiers(Modifier.PUBLIC)
                .addException(Exception.class)
                .addJavadoc("Swaps every delegate for a test double; restore() afterwards is mandatory.\n");

        ClassName facadeName = ClassName.get(facade.facadeClass());
        if (facade.needsInstance()) {
            b.addStatement("$T target = $T.$L($L)", facadeName, facadeName,
                    facade.accessor().getName(), accessorArgs(facade));
        }
        for (SMCFixtureScanner.DelegateField d : facade.delegates()) {
            String fieldVar = d.name() + "Field";
            b.addStatement("$T $L = $T.class.getDeclaredField($S)",
                    REFLECT_FIELD, fieldVar, facadeName, d.name());
            b.addStatement("$L.setAccessible(true)", fieldVar);
            b.addStatement("originals.put($S, $L.get($L))", d.name(), fieldVar, targetOf(d));
            if (d.mapShaped()) {
                b.addStatement("$L = new $T<>(($T<?, ?>) originals.get($S))",
                        handleName(d), LINKED_MAP, MAP, d.name());
                for (String key : seededKeys(facade)) {
                    b.addStatement("$L.put($S, $S)", handleName(d), key,
                            com.config.Config.SEEDED_PLACEHOLDER_VALUE);
                }
            } else if (seedable(d)) {
                b.addStatement("$L = new $T()", handleName(d), rawType(d));
                for (String key : seededKeys(facade)) {
                    b.addStatement("$L.setProperty($S, $S)", handleName(d), key,
                            com.config.Config.SEEDED_PLACEHOLDER_VALUE);
                }
            } else {
                b.addStatement("$L = $T.mock($T.class)", handleName(d), MOCKITO, rawType(d));
            }
            b.addStatement("$L.set($L, $L)", fieldVar, targetOf(d), handleName(d));
        }
        return b.build();
    }

    /** restore(): put every original back — the delegates are JVM-global state. */
    private static MethodSpec restoreMethod(SMCFixtureScanner.Facade facade) {
        MethodSpec.Builder b = MethodSpec.methodBuilder("restore")
                .addModifiers(Modifier.PUBLIC)
                .addException(Exception.class);

        ClassName facadeName = ClassName.get(facade.facadeClass());
        if (facade.needsInstance()) {
            b.addStatement("$T target = $T.$L($L)", facadeName, facadeName,
                    facade.accessor().getName(), accessorArgs(facade));
        }
        for (SMCFixtureScanner.DelegateField d : facade.delegates()) {
            String fieldVar = d.name() + "Field";
            b.beginControlFlow("if (originals.containsKey($S))", d.name());
            b.addStatement("$T $L = $T.class.getDeclaredField($S)",
                    REFLECT_FIELD, fieldVar, facadeName, d.name());
            b.addStatement("$L.setAccessible(true)", fieldVar);
            b.addStatement("$L.set($L, originals.get($S))", fieldVar, targetOf(d), d.name());
            b.endControlFlow();
        }
        return b.build();
    }

    /** Map delegates keep their runtime shape via a live LinkedHashMap copy; others get a mock. */
    private static com.squareup.javapoet.TypeName handleType(SMCFixtureScanner.DelegateField d) {
        if (d.mapShaped()) {
            return ParameterizedTypeName.get(LINKED_MAP,
                    ClassName.get(Object.class), ClassName.get(Object.class));
        }
        return ClassName.get(d.type());
    }

    private static ClassName rawType(SMCFixtureScanner.DelegateField d) {
        return ClassName.get(d.type());
    }

    private static String handleName(SMCFixtureScanner.DelegateField d) {
        return d.name() + (d.mapShaped() ? "Values" : "Mock");
    }

    private static String targetOf(SMCFixtureScanner.DelegateField d) {
        return d.staticField() ? "null" : "target";
    }

    /** Class params get the fixture's own class literal; String params a fixed token. */
    private static CodeBlock accessorArgs(SMCFixtureScanner.Facade facade) {
        CodeBlock.Builder args = CodeBlock.builder();
        boolean first = true;
        for (Class<?> p : facade.accessor().getParameterTypes()) {
            if (!first) {
                args.add(", ");
            }
            first = false;
            if (p == Class.class) {
                args.add("$L.class", fixtureSimpleName(facade));
            } else {
                args.add("$S", "testgen");
            }
        }
        return args.build();
    }

    /**
     * TWEAK-POINT (CONFIG_PAYLOAD=SEEDED): a delegate is seedable when it has a public no-arg
     * constructor and a setProperty(String, Object) seam. Anything else falls back to a mock.
     */
    private static boolean seedable(SMCFixtureScanner.DelegateField d) {
        if (!"SEEDED".equals(com.config.Config.CONFIG_PAYLOAD)) {
            return false;
        }
        try {
            d.type().getConstructor();
            d.type().getMethod("setProperty", String.class, Object.class);
            return true;
        } catch (Throwable t) {
            return false;
        }
    }

    /**
     * TWEAK-POINT (CONFIG_PAYLOAD=SEEDED): key discovery. Scans every main-source file for string
     * literals on lines that mention the facade's simple name or its accessor, keeping literals
     * that look like config keys (dotted or underscored, no spaces). Over-collection is harmless -
     * an extra key just carries the placeholder value.
     */
    private static java.util.List<String> seededKeys(SMCFixtureScanner.Facade facade) {
        if (!"SEEDED".equals(com.config.Config.CONFIG_PAYLOAD)) {
            return java.util.List.of();
        }
        java.util.regex.Pattern literal = java.util.regex.Pattern.compile("\"([A-Za-z0-9_.]*[._][A-Za-z0-9_.]*)\"");
        java.util.LinkedHashSet<String> keys = new java.util.LinkedHashSet<>();
        String simple = facade.facadeClass().getSimpleName();
        String accessor = facade.accessor().getName();
        try (var files = java.nio.file.Files.walk(com.config.Config.mainSourceRoot())) {
            for (java.nio.file.Path p : files.filter(f -> f.toString().endsWith(".java")).toList()) {
                for (String line : java.nio.file.Files.readAllLines(p)) {
                    if (!line.contains(simple) && !line.contains(accessor + "(")) {
                        continue;
                    }
                    java.util.regex.Matcher m = literal.matcher(line);
                    while (m.find()) {
                        keys.add(m.group(1));
                    }
                }
            }
        } catch (Throwable ignored) { }
        return new java.util.ArrayList<>(keys);
    }

/**
 * Adds the facade protection (install/restore + fixture) to EXISTING test classes that the
 * generator did not produce. Deterministic and anchor-based: when an expected anchor is missing
 * the file is reported and SKIPPED, never guessed at or corrupted.
 *
 * Honours the same switches as generation:
 *   Config.FIXTURE_PLACEMENT  "SHARED" = testsupport fixture files + thin calls
 *                             "NESTED" = fixture embedded privately in each injected file
 *   Config.CONFIG_PAYLOAD     "MOCK" | "SEEDED" (see FixtureGenerator)
 *
 * Run from the module root (IntelliJ Run works): reads testgen/classes.txt like the Runner.
 */
public final class InjectSMCFixture {

    private InjectSMCFixture() {}

    public static void main(String[] args) throws Exception {
        for (String fqcn : Config.readClasses()) {
            Class<?> sut = Class.forName(fqcn.trim());
            List<SMCFixtureScanner.Facade> facades = SMCFixtureScanner.detect(sut);
            if (facades.isEmpty()) {
                System.out.println("SKIP (no facade detected): " + fqcn);
                continue;
            }
            Path testFile = Config.testSourceRoot()
                    .resolve(sut.getPackageName().replace('.', '/'))
                    .resolve(sut.getSimpleName() + Config.TEST_NAME_SUFFIX + ".java");
            if (!Files.isRegularFile(testFile)) {
                System.out.println("SKIP (no existing test file): " + testFile);
                continue;
            }
            String source = Files.readString(testFile);
            if (source.contains("Fixture")) {
                System.out.println("SKIP (already protected): " + testFile);
                continue;
            }
            String injected = inject(source, sut, facades, testFile);
            if (injected == null) {
                continue;   // anchor missing - reported inside inject()
            }
            Files.writeString(testFile, injected);
            System.out.println("INJECTED " + facades.size() + " fixture(s): " + testFile);
        }
    }

    /**
     * TWEAK-POINT (injection anchors): the class-body anchor is the first "class <TestName> {"
     * line; setup/tearDown anchors are "void setup()" / "void tearDown()" if present, else both
     * methods are created after the fixture fields.
     */
    private static String inject(String source, Class<?> sut,
            List<SMCFixtureScanner.Facade> facades, Path testFile) throws Exception {
        String testName = sut.getSimpleName() + Config.TEST_NAME_SUFFIX;
        String classAnchor = "class " + testName + " {";
        int classIdx = source.indexOf(classAnchor);
        if (classIdx < 0) {
            System.out.println("SKIP (class anchor missing): " + testFile);
            return null;
        }
        int bodyStart = classIdx + classAnchor.length();

        StringBuilder fields = new StringBuilder();
        StringBuilder installs = new StringBuilder();
        StringBuilder restores = new StringBuilder();
        StringBuilder nestedTypes = new StringBuilder();
        StringBuilder imports = new StringBuilder();

        for (SMCFixtureScanner.Facade facade : facades) {
            String simple = FixtureGenerator.fixtureSimpleName(facade);
            String var = Character.toLowerCase(simple.charAt(0)) + simple.substring(1);
            fields.append("\n    private final ").append(simple).append(' ').append(var)
                    .append(" = new ").append(simple).append("();\n");
            installs.append("        ").append(var).append(".install();\n");
            restores.insert(0, "        " + var + ".restore();\n");
            if ("NESTED".equals(Config.FIXTURE_PLACEMENT)) {
                String rendered = renderedNested(facade);
                nestedTypes.append('\n').append(indent(classOf(rendered))).append('\n');
                imports.append(importsOf(rendered));
            } else {
                FixtureGenerator.write(facade, Config.testSourceRoot());
                imports.append("import testsupport.").append(simple).append(";\n");
            }
        }

        String out = source;
        if (imports.length() > 0) {
            int pkgEnd = out.indexOf(';', out.indexOf("package ")) + 1;
            out = out.substring(0, pkgEnd) + "\n\n" + imports + out.substring(pkgEnd);
            bodyStart = out.indexOf(classAnchor) + classAnchor.length();
        }

        boolean hasSetup = out.contains("void setup()");
        boolean hasTearDown = out.contains("void tearDown()");
        String lifecycle = "";
        if (!hasSetup) {
            lifecycle += "\n    @org.junit.jupiter.api.BeforeEach\n    void setup() throws Exception {\n"
                    + installs + "    }\n";
        }
        if (!hasTearDown) {
            lifecycle += "\n    @org.junit.jupiter.api.AfterEach\n    void tearDown() throws Exception {\n"
                    + restores + "    }\n";
        }
        out = out.substring(0, bodyStart) + fields + lifecycle + out.substring(bodyStart);

        if (hasSetup) {
            out = insertAtMethodStart(out, "void setup()", installs.toString(), testFile);
            if (out == null) {
                return null;
            }
        }
        if (hasTearDown) {
            out = insertAtMethodStart(out, "void tearDown()", restores.toString(), testFile);
            if (out == null) {
                return null;
            }
        }

        if (nestedTypes.length() > 0) {
            int lastBrace = out.lastIndexOf('}');
            out = out.substring(0, lastBrace) + nestedTypes + "}\n";
        }
        return out;
    }

    private static String insertAtMethodStart(String source, String signature, String lines,
            Path testFile) {
        int idx = source.indexOf(signature);
        int brace = source.indexOf('{', idx);
        if (idx < 0 || brace < 0) {
            System.out.println("SKIP (lifecycle anchor missing '" + signature + "'): " + testFile);
            return null;
        }
        return source.substring(0, brace + 1) + "\n" + lines + source.substring(brace + 1);
    }

    /** The nested fixture rendered as a full compilation unit (imports + class). */
    private static String renderedNested(SMCFixtureScanner.Facade facade) {
        TypeSpec type = FixtureGenerator.nestedType(facade);
        return JavaFile.builder("injected", type).skipJavaLangImports(true).build().toString();
    }

    private static String classOf(String rendered) {
        return rendered.substring(rendered.indexOf("private static final class"));
    }

    /** The fixture's own import lines, so the host file compiles without JavaPoet's help. */
    private static String importsOf(String rendered) {
        StringBuilder out = new StringBuilder();
        for (String line : rendered.split("\n")) {
            if (line.startsWith("import ")) {
                out.append(line).append('\n');
            }
        }
        return out.toString();
    }

    private static String indent(String block) {
        StringBuilder out = new StringBuilder();
        for (String line : block.split("\n")) {
            out.append(line.isBlank() ? "" : "    " + line).append('\n');
        }
        return out.toString();
    }
}

/**
 * Flips EXISTING generated tests between fixture placements for the logger/config fixtures.
 *
 * SHARED -> NESTED: drops the testsupport import and embeds the rendered fixture as a private
 * nested class. NESTED -> SHARED: writes the testsupport fixture (skip-if-exists), removes the

 *

 * blank runs are capped so repeated flips cannot grow the file.
 *
 * Anchor-based and skip-not-corrupt: an unexpected file is reported and left alone. Idempotent:
 * a file already in the target placement is a SKIP.
 *
 * Usage: run main with optional arg SHARED|NESTED; no arg = Config.FIXTURE_PLACEMENT.
 */

    private SwitchSMCFixture() {}

    public static void main(String[] args) throws Exception {

        if (!target.equals("SHARED") && !target.equals("NESTED")) {
            throw new IllegalArgumentException("target must be SHARED or NESTED, got: " + target);
        }

        for (String fqcn : Config.readClasses()) {
            Class<?> sut = Class.forName(fqcn.trim());
            List<SMCFixtureScanner.Facade> facades = SMCFixtureScanner.detect(sut);
            if (facades.isEmpty()) {
                System.out.println("SKIP (no facade detected): " + fqcn);
                continue;
            }

            Path testFile = Config.testSourceRoot()
                    .resolve(sut.getPackageName().replace('.', '/'))
                    .resolve(sut.getSimpleName() + Config.TEST_NAME_SUFFIX + ".java");
            if (!Files.isRegularFile(testFile)) {
                System.out.println("SKIP (no existing test file): " + testFile);
                continue;
            }

            migrate(testFile, facades, target);
        }
    }

    /** Migrates one test file to the target placement; counts per-facade outcomes. */
    private static void migrate(Path testFile, List<SMCFixtureScanner.Facade> facades,
            String target) throws Exception {
        String source = Files.readString(testFile);
        int changed = 0;

        for (SMCFixtureScanner.Facade facade : facades) {
            String simple = FixtureGenerator.fixtureSimpleName(facade);
            boolean nestedHere = source.contains("private static final class " + simple);
            boolean sharedHere = source.contains("import testsupport." + simple + ";");

            if (target.equals("NESTED") && sharedHere && !nestedHere) {
                source = toNested(source, facade, simple, testFile);
                if (source == null) {
                    return;
                }
                changed++;
            } else if (target.equals("SHARED") && nestedHere) {
                source = toShared(source, facade, simple, testFile);
                if (source == null) {
                    return;
                }
                changed++;
            }
        }

        if (changed == 0) {
            System.out.println("SKIP (already " + target + "): " + testFile);
            return;
        }
        Files.writeString(testFile, capBlankRuns(source));
        System.out.println("MIGRATED " + changed + " fixture(s) -> " + target + ": " + testFile);
    }

    /** SHARED -> NESTED: remove the import, embed the rendered fixture before the final brace. */
    private static String toNested(String source, SMCFixtureScanner.Facade facade,
            String simple, Path testFile) {
        String importLine = "import testsupport." + simple + ";";
        int importIdx = source.indexOf(importLine);
        int lastBrace = source.lastIndexOf('}');
        if (importIdx < 0 || lastBrace < 0) {
            System.out.println("SKIP (anchor missing for " + simple + "): " + testFile);
            return null;
        }

        String rendered = renderNested(facade);
        String out = source.replace(importLine + "\n", "").replace(importLine, "");
        out = mergeImports(out, fixtureImports(rendered, hostPackage(out)), testFile);
        if (out == null) {
            return null;
        }

        lastBrace = out.lastIndexOf('}');
        return out.substring(0, lastBrace) + "\n" + indent(classOf(rendered)) + "}\n";
    }

    /** NESTED -> SHARED: write the testsupport fixture, cut the nested block, add the import. */
    private static String toShared(String source, SMCFixtureScanner.Facade facade,
            String simple, Path testFile) throws Exception {
        String anchor = "private static final class " + simple;
        int start = source.indexOf(anchor);
        int open = source.indexOf('{', start);
        if (start < 0 || open < 0) {
            System.out.println("SKIP (nested anchor missing for " + simple + "): " + testFile);
            return null;
        }

        int depth = 0;
        int end = -1;
        for (int i = open; i < source.length(); i++) {
            char c = source.charAt(i);
            if (c == '{') {
                depth++;
            } else if (c == '}') {
                depth--;
                if (depth == 0) {
                    end = i + 1;
                    break;
                }
            }
        }
        if (end < 0) {
            System.out.println("SKIP (unbalanced braces in " + simple + "): " + testFile);
            return null;
        }

        FixtureGenerator.write(facade, Config.testSourceRoot());

        int lineStart = source.lastIndexOf('\n', start) + 1;
        String out = source.substring(0, lineStart) + source.substring(end);
        out = mergeImports(out, "import testsupport." + simple + ";\n", testFile);
        if (out == null) {
            return null;
        }
        return dropUnusedImports(out);
    }

    /** Inserts import lines (deduplicated) after the package statement. */
    private static String mergeImports(String source, String imports, Path testFile) {
        int pkgIdx = source.indexOf("package ");
        int pkgEnd = source.indexOf(';', pkgIdx);
        if (pkgIdx < 0 || pkgEnd < 0) {
            System.out.println("SKIP (package anchor missing): " + testFile);
            return null;
        }

        StringBuilder fresh = new StringBuilder();
        for (String line : imports.split("\n")) {
            if (!line.isBlank() && !source.contains(line.trim())) {
                fresh.append(line.trim()).append('\n');
            }
        }
        if (fresh.length() == 0) {
            return source;
        }

        if (source.indexOf("\nimport ") < 0) {
            return source.substring(0, pkgEnd + 1) + "\n\n" + fresh + source.substring(pkgEnd + 1);
        }
        for (String add : fresh.toString().split("\n")) {
            int insertAt = sortedInsertPoint(source, add);
            source = source.substring(0, insertAt) + add + "\n" + source.substring(insertAt);
        }
        return source;
    }


    /** R3: position keeping the import block alphabetically ordered. */
    private static int sortedInsertPoint(String source, String importLine) {
        int idx = source.indexOf("\nimport ");
        int insertAt = -1;
        int lastEnd = -1;
        while (idx >= 0) {
            int lineEnd = source.indexOf('\n', idx + 1);
            String existing = source.substring(idx + 1, lineEnd);
            if (insertAt < 0 && existing.compareTo(importLine) > 0) {
                insertAt = idx + 1;
            }
            lastEnd = lineEnd + 1;
            idx = source.indexOf("\nimport ", idx + 1);
        }
        return insertAt >= 0 ? insertAt : lastEnd;
    }

    /** D1: the fixture's import lines, filtered - java.lang and the host package never enter. */
    private static String fixtureImports(String rendered, String hostPackage) {
        StringBuilder out = new StringBuilder();
        for (String line : rendered.split("\n")) {
            if (!line.startsWith("import ")) {
                continue;
            }
            String fqcn = line.substring(7).replace(";", "").trim();
            String pkg = fqcn.contains(".") ? fqcn.substring(0, fqcn.lastIndexOf('.')) : "";
            if (pkg.equals("java.lang") || pkg.equals(hostPackage)) {
                continue;
            }
            out.append(line).append('\n');
        }
        return out.toString();
    }

    /** The host file's own package, for import filtering. */
    private static String hostPackage(String source) {
        int idx = source.indexOf("package ");
        int endIdx = source.indexOf(';', idx);
        return idx < 0 ? "" : source.substring(idx + 8, endIdx).trim();
    }

    /** D1: after a flip to SHARED, imports whose simple name no longer occurs are dropped. */
    private static String dropUnusedImports(String source) {
        StringBuilder out = new StringBuilder();
        for (String line : source.split("\n", -1)) {
            String trimmed = line.trim();
            if (trimmed.startsWith("import ") && trimmed.endsWith(";")
                    && !trimmed.startsWith("import static")) {
                String fqcn = trimmed.substring(7, trimmed.length() - 1).trim();
                if (fqcn.startsWith("java.lang.") && fqcn.chars().filter(c -> c == '.').count() == 2) {
                    continue;   // W1 residue: java.lang imports are always redundant
                }
                String simple = fqcn.substring(fqcn.lastIndexOf('.') + 1);
                String body = source.substring(source.indexOf(line) + line.length());
                java.util.regex.Pattern usage = java.util.regex.Pattern.compile(
                        "\\b" + java.util.regex.Pattern.quote(simple) + "\\b");
                if (!usage.matcher(body).find()) {
                    continue;
                }
            }
            out.append(line).append('\n');
        }
        String result = out.toString();
        return result.substring(0, result.length() - 1);
    }

    /** D2: flips must not grow whitespace - blank runs are capped at two. */
    private static String capBlankRuns(String source) {
        String unified = source.replace("\r\n", "\n");
        return unified.replaceAll("\n{4,}", "\n\n\n");
    }

    /** The fixture rendered as a full compilation unit (imports + class). */
    private static String renderNested(SMCFixtureScanner.Facade facade) {
        TypeSpec type = FixtureGenerator.nestedType(facade);
        return JavaFile.builder("migrated", type).skipJavaLangImports(true).build().toString();
    }

    private static String classOf(String rendered) {
        return rendered.substring(rendered.indexOf("private static final class"));
    }

    private static String indent(String block) {
        StringBuilder out = new StringBuilder();
        for (String line : block.split("\n")) {
            out.append(line.isBlank() ? "" : "    " + line).append('\n');
        }
        return out.toString();
    }
}

class SkipFile(Exception):
    """Raised when a file does not match expectations; the file is left alone."""


class JavaFile:
    """A test file as lines, with its newline style preserved end to end."""

    def __init__(self, path):
        self.path = path
        raw = path.read_bytes()
        try:
            text = raw.decode('utf-8')
        except UnicodeDecodeError:
            raise SkipFile('not valid UTF-8')
        self.eol = self._detect_eol(text)
        self.ends_with_newline = text.endswith(('\n', '\r\n'))
        self.original_text = text
        self.lines = text.replace('\r\n', '\n').split('\n')
        if self.ends_with_newline:
            self.lines.pop()

    @staticmethod
    def _detect_eol(text):
        crlf = text.count('\r\n')
        if text.count('\r') != crlf:
            raise SkipFile('mixed or bare-CR line endings')
        if crlf and crlf != text.count('\n'):
            raise SkipFile('mixed CRLF/LF line endings')
        return '\r\n' if crlf else '\n'

    def render(self):
        text = self.eol.join(self.lines)
        return text + self.eol if self.ends_with_newline else text

    def changed(self):
        return self.render() != self.original_text


def strip_comments_and_strings(text):
    """Blanks out //, /* */ comments and string/char literals for brace counting."""
    out = []
    i, n, state = 0, len(text), None
    while i < n:
        c = text[i]
        nxt = text[i + 1] if i + 1 < n else ''
        if state is None:
            if c == '/' and nxt == '/':
                state = 'line'
                i += 2
                continue
            if c == '/' and nxt == '*':
                state = 'block'
                i += 2
                continue
            if c in ('"', "'"):
                state = c
                i += 1
                continue
            out.append(c)
        elif state == 'line':
            if c == '\n':
                state = None
                out.append(c)
        elif state == 'block':
            if c == '*' and nxt == '/':
                state = None
                i += 1
            elif c == '\n':
                out.append(c)
        else:
            if c == '\\':
                i += 1
            elif c == state:
                state = None
            elif c == '\n':
                state = None    # unterminated literal: fail safe at line end
                out.append(c)
        i += 1
    return ''.join(out)


def check_brace_balance(java):
    """Whole-file brace sanity; a mismatch means we must not touch the file."""
    stripped = strip_comments_and_strings('\n'.join(java.lines))
    depth = 0
    for c in stripped:
        if c == '{':
            depth += 1
        elif c == '}':
            depth -= 1
        if depth < 0:
            raise SkipFile('unbalanced braces')
    if depth != 0:
        raise SkipFile('unbalanced braces')


def code_lines(java):
    """Per-line comment/string-stripped view used for anchors and depth scans."""
    return strip_comments_and_strings('\n'.join(java.lines)).split('\n')


def find_class_block(java, test_simple_name):
    """Returns (open_idx, close_idx) of the test class declaration and its brace."""
    pattern = re.compile(
        r'^\s*(public\s+)?(final\s+)?class\s+%s\b.*\{\s*$' % re.escape(test_simple_name))
    stripped = code_lines(java)
    open_idx = next((i for i, line in enumerate(stripped) if pattern.match(line)), None)
    if open_idx is None:
        raise SkipFile('class anchor "class %s {" not found' % test_simple_name)
    close_idx = matching_close(stripped, open_idx)
    if close_idx is None:
        raise SkipFile('class closing brace not found')
    return open_idx, close_idx


def matching_close(stripped, open_idx):
    """Line index whose brace returns the depth opened at open_idx to zero."""
    depth = 0
    for i in range(open_idx, len(stripped)):
        depth += stripped[i].count('{') - stripped[i].count('}')
        if depth == 0:
            return i
    return None


def find_lifecycle(java, annotation, p48_name):
    """
    Finds the @BeforeEach/@AfterEach method; enforces the P48 name.

    Returns (annotation_idx, signature_idx, close_idx) or None when absent.
    """
    hits = [i for i, line in enumerate(java.lines) if line.strip() == annotation]
    if not hits:
        return None
    if len(hits) > 1:
        raise SkipFile('multiple %s methods' % annotation)
    ann_idx = hits[0]
    sig_idx = next((i for i in range(ann_idx + 1, len(java.lines))
                    if java.lines[i].strip()), None)
    sig = re.match(r'^\s*public\s+void\s+(\w+)\s*\([^)]*\)[^{]*\{\s*$',
                   java.lines[sig_idx] if sig_idx is not None else '')
    if not sig:
        raise SkipFile('%s not followed by a plain method signature' % annotation)
    if sig.group(1) != p48_name:
        raise SkipFile('%s method is named %s, expected %s'
                       % (annotation, sig.group(1), p48_name))
    close_idx = matching_close(code_lines(java), sig_idx)
    if close_idx is None:
        raise SkipFile('%s method has no closing brace' % p48_name)
    return ann_idx, sig_idx, close_idx


def member_indent(java, class_open):
    """Indentation of class members, taken from the first member line found."""
    for line in java.lines[class_open + 1:]:
        if line.strip():
            return line[:len(line) - len(line.lstrip())] or '  '
    return '  '


def host_package(java):
    for line in java.lines:
        match = re.match(r'^\s*package\s+([\w.]+)\s*;', line)
        if match:
            return match.group(1)
    raise SkipFile('package declaration not found')


def word_used_outside_imports(java, word):
    """True when the simple name occurs anywhere except import lines."""
    body = '\n'.join(line for line in java.lines
                     if not line.strip().startswith('import '))
    return re.search(r'\b%s\b' % re.escape(word), body) is not None


def add_import(java, import_line):
    """Inserts an import in alphabetical order; no-op when already present."""
    if any(line.strip() == import_line for line in java.lines):
        return
    import_idxs = [i for i, line in enumerate(java.lines)
                   if line.strip().startswith('import ')]
    if import_idxs:
        after = [i for i in import_idxs if java.lines[i].strip() > import_line]
        at = after[0] if after else import_idxs[-1] + 1
    else:
        pkg_idx = next(i for i, line in enumerate(java.lines)
                       if line.strip().startswith('package '))
        java.lines.insert(pkg_idx + 1, '')
        at = pkg_idx + 2
    java.lines.insert(at, import_line)


def remove_import_if_unused(java, import_line):
    """Drops an import the tool may have added, when its name is no longer used."""
    simple = import_line.rstrip(';').split('.')[-1]
    for i, line in enumerate(java.lines):
        if line.strip() == import_line:
            if not word_used_outside_imports(java, simple):
                del java.lines[i]
            return


def cap_blank_runs(java):
    """Edits must never grow whitespace: blank runs are capped at two lines."""
    out, run = [], 0
    for line in java.lines:
        run = run + 1 if not line.strip() else 0
        if run <= 2:
            out.append(line)
    java.lines = out


def fixture_presence(java, fixture):
    """Returns 'shared', 'nested', 'wired' (lines only), or None."""
    text = '\n'.join(java.lines)
    if fixture.shared_import in text:
        return 'shared'
    if fixture.nested_anchor in text:
        return 'nested'
    if fixture.field_decl in text or fixture.install_line in text:
        return 'wired'
    return None


def check_no_name_conflict(java, fixture):
    """A same-named member that is not our exact wiring means hands off."""
    if word_used_outside_imports(java, fixture.field_name):
        raise SkipFile('field name %s already in use' % fixture.field_name)
    if word_used_outside_imports(java, fixture.simple):
        raise SkipFile('type name %s already in use' % fixture.simple)

def ensure_lifecycle_methods(java, class_open, class_close, indent):
    """Creates setup()/tearDown() with P48 names when absent; returns nothing."""
    if find_lifecycle(java, '@BeforeEach', 'setup') is None:

        java.lines[at:at] = block
        add_import(java, IMPORT_BEFORE_EACH)
    if find_lifecycle(java, '@AfterEach', 'tearDown') is None:
        _, _, setup_close = find_lifecycle(java, '@BeforeEach', 'setup')
        at = setup_close + 1
        if at < len(java.lines) and not java.lines[at].strip():
            at += 1
        block = [indent + '@AfterEach',
                 indent + 'public void tearDown() throws Exception {',
                 indent + '}',
                 '']
        java.lines[at:at] = block
        add_import(java, IMPORT_AFTER_EACH)

    for i in range(class_open + 1, class_close):

            return i
    return class_close

            break
        earlier = FIXTURES[name]
        if at < len(java.lines) and java.lines[at].strip() == earlier.field_decl:
            at += 1
            if at < len(java.lines) and not java.lines[at].strip():
                at += 1
    java.lines[at:at] = [indent + fixture.field_decl, '']


def insert_install(java, fixture, indent):
    """Install goes at the top of setup(); logger installs before config ones."""
    _, sig_idx, close_idx = find_lifecycle(java, '@BeforeEach', 'setup')
    body_indent = indent * 2
    at = sig_idx + 1
    if fixture.kind == 'config':
        at = after_last_known_line(java, sig_idx, close_idx,
                                   known_install_lines('logger'), at)
    java.lines.insert(at, body_indent + fixture.install_line)


def insert_restore(java, fixture, indent):
    """Restore order mirrors install in reverse: config restores before logger."""
    _, sig_idx, close_idx = find_lifecycle(java, '@AfterEach', 'tearDown')
    body_indent = indent * 2
    at = sig_idx + 1
    if fixture.kind == 'logger':
        at = after_last_known_line(java, sig_idx, close_idx,
                                   known_restore_lines('config'), at)
    java.lines.insert(at, body_indent + fixture.restore_line)


def known_install_lines(kind):
    return [f.install_line for f in FIXTURES.values() if f.kind == kind]


def known_restore_lines(kind):
    return [f.restore_line for f in FIXTURES.values() if f.kind == kind]


def after_last_known_line(java, sig_idx, close_idx, known, default):
    """Index just past the last known line inside the method body, else default."""
    at = default
    for i in range(sig_idx + 1, close_idx):
        if java.lines[i].strip() in known:
            at = i + 1
    return at


def inject(java, fixture, placement, test_root):
    """Injects one fixture (shared or nested placement) into the file."""
    presence = fixture_presence(java, fixture)
    if presence:
        raise SkipFile('%s already present (%s)' % (fixture.simple, presence))
    check_no_name_conflict(java, fixture)

    if placement == 'shared':
        support = test_root / 'testsupport' / (fixture.simple + '.java')
        if not support.is_file():
            raise SkipFile('missing %s' % support)

    test_name = Path(java.path).stem
    class_open, class_close = find_class_block(java, test_name)
    indent = member_indent(java, class_open)
    pkg = host_package(java)

    if placement == 'shared':
        add_import(java, fixture.shared_import)
    else:
        block = fixture.nested_text.split('\n')
        if java.lines[class_close - 1].strip():
            block = [''] + block
        java.lines[class_close:class_close] = block
        for fqcn in fixture.nested_imports:
            if fqcn.rsplit('.', 1)[0] != pkg:
                add_import(java, 'import %s;' % fqcn)

    class_open, class_close = find_class_block(java, test_name)
    insert_field(java, fixture, class_open, indent)
    class_open, class_close = find_class_block(java, test_name)
    ensure_lifecycle_methods(java, class_open, class_close, indent)
    insert_install(java, fixture, indent)
    insert_restore(java, fixture, indent)


# ------------------------------------------------------------------- removal


def remove_line(java, content):
    """Removes the first line with this stripped content; True when found."""
    for i, line in enumerate(java.lines):
        if line.strip() == content:
            del java.lines[i]
            return True
    return False


def remove_field(java, fixture):
    """Removes the field line plus the single blank separator the tool adds."""
    for i, line in enumerate(java.lines):
        if line.strip() == fixture.field_decl:
            del java.lines[i]
            if i < len(java.lines) and not java.lines[i].strip():
                del java.lines[i]
            return


def remove_nested_block(java, fixture):
    """
    Cuts the embedded nested class block, located by its anchor line.

    Inject adds a separating blank line only when the block would touch the
    previous member; removal mirrors that exactly — the single preceding blank
    goes only when its own predecessor is non-blank. Without this pairing a
    nested round trip on a generator-shaped file would grow the file.
    """
    stripped = code_lines(java)
    start = next((i for i, line in enumerate(java.lines)
                  if line.strip().startswith(fixture.nested_anchor)), None)
    if start is None:
        raise SkipFile('nested anchor missing for %s' % fixture.simple)
    end = matching_close(stripped, start)
    if end is None:
        raise SkipFile('nested block unbalanced for %s' % fixture.simple)
    del java.lines[start:end + 1]
    if (start > 1 and not java.lines[start - 1].strip()
            and java.lines[start - 2].strip()):
        del java.lines[start - 1]


def remove_empty_lifecycle(java, annotation, p48_name):
    """Drops a lifecycle method the removal emptied, plus its trailing blank."""
    found = find_lifecycle(java, annotation, p48_name)
    if found is None:
        return
    ann_idx, sig_idx, close_idx = found
    body = [line for line in java.lines[sig_idx + 1:close_idx] if line.strip()]
    if body:
        return
    del java.lines[ann_idx:close_idx + 1]
    if ann_idx < len(java.lines) and not java.lines[ann_idx].strip():
        del java.lines[ann_idx]
    import_line = (IMPORT_BEFORE_EACH if annotation == '@BeforeEach'
                   else IMPORT_AFTER_EACH)
    remove_import_if_unused(java, import_line)


def remove(java, fixture, placement):
    """Removes one fixture's wiring; strictly the reverse of inject."""
    presence = fixture_presence(java, fixture)
    if presence is None:
        raise SkipFile('%s not present' % fixture.simple)
    if presence == 'wired':
        raise SkipFile('%s wiring present without import or nested class'
                       % fixture.simple)
    if presence != placement:
        raise SkipFile('%s present as %s; use remove-%s'
                       % (fixture.simple, presence, presence))

    if placement == 'shared':
        remove_line(java, fixture.shared_import)
    else:
        remove_nested_block(java, fixture)

    remove_field(java, fixture)
    remove_line(java, fixture.install_line)
    remove_line(java, fixture.restore_line)
    remove_empty_lifecycle(java, '@BeforeEach', 'setup')
    remove_empty_lifecycle(java, '@AfterEach', 'tearDown')

    if placement == 'nested':
        for fqcn in fixture.nested_imports:
            remove_import_if_unused(java, 'import %s;' % fqcn)


# ------------------------------------------------------------------ resolve


def read_classes(classes_file):
    """SUT class names, one per line; blanks and # comments ignored."""
    names = []
    for line in classes_file.read_text(encoding='utf-8').splitlines():
        line = line.strip()
        if line and not line.startswith('#'):
            names.append(line)
    return names


def test_file_for(test_root, fqcn):
    parts = fqcn.split('.')
    return test_root.joinpath(*parts[:-1], parts[-1] + 'Test.java')


def fixtures_for_inject(fqcn, kinds):
    """Registry lookup, no detection: which fixtures inject for this SUT."""
    chosen = []
    if 'logger' in kinds:
        pkg = fqcn.rsplit('.', 1)[0]
        name = next((fix for prefix, fix in LOGGER_BY_PACKAGE
                     if pkg == prefix or pkg.startswith(prefix + '.')), None)
        if name is None:
            raise SkipFile('no logger fixture mapping for package of %s' % fqcn)
        chosen.append(FIXTURES[name])
    if 'config' in kinds:
        chosen.append(FIXTURES['SettingsHubFixture'])
    return in_canonical_order(chosen)


def fixtures_for_remove(java, kinds):
    """Whichever known fixtures of the requested kinds are actually in the file."""
    present = [f for f in FIXTURES.values()
               if f.kind in kinds and fixture_presence(java, f)]
    if not present:
        raise SkipFile('no %s fixture present' % '/'.join(sorted(kinds)))
    return in_canonical_order(present)


def in_canonical_order(fixtures):
    return sorted(fixtures, key=lambda f: CANONICAL_ORDER.index(f.simple))


# --------------------------------------------------------------------- main


def process_class(fqcn, args, test_root):
    """Runs the action for one SUT class; returns (status, detail) for the report."""
    path = test_file_for(test_root, fqcn)
    if not path.is_file():
        return 'SKIP', 'test file not found: %s' % path

    java = JavaFile(path)
    check_brace_balance(java)
    find_class_block(java, path.stem)

    mode, placement = args.action.split('-')
    kinds = {'both': {'logger', 'config'}}.get(args.fixture, {args.fixture})

    if mode == 'inject':
        chosen = fixtures_for_inject(fqcn, kinds)
    else:
        chosen = fixtures_for_remove(java, kinds)

    applied, skipped = [], []
    for fixture in chosen:
        before = list(java.lines)
        try:
            if mode == 'inject':
                inject(java, fixture, placement, test_root)
            else:
                remove(java, fixture, placement)
            applied.append(fixture.simple)
        except SkipFile as reason:
            java.lines = before
            skipped.append('%s (%s)' % (fixture.simple, reason))

    if not applied:
        return 'SKIP', '; '.join(skipped) + ' -> %s' % path

    cap_blank_runs(java)
    detail = '%s -> %s' % ('+'.join(applied), path)
    if skipped:
        detail += ' [skipped: %s]' % '; '.join(skipped)
    if args.dry_run:
        diff = difflib.unified_diff(
            java.original_text.splitlines(), java.render().splitlines(),
            fromfile=str(path), tofile='%s (%s)' % (path, args.action), lineterm='')
        print('\n'.join(diff))
        return 'DRY-RUN', detail
    path.write_bytes(java.render().encode('utf-8'))
    return mode.upper(), detail


def parse_args(argv):
    parser = argparse.ArgumentParser(
        prog='fixture_housekeeper',
        description='Inject or remove testgen logger/config fixture boilerplate.')
    parser.add_argument('--classes', required=True, type=Path,
                        help='file with one fully-qualified SUT class per line')
    parser.add_argument('--test-root', required=True, type=Path,
                        help='test source root, e.g. <project>/src/test/java')
    parser.add_argument('--action', required=True,
                        choices=['inject-nested', 'remove-nested',
                                 'inject-shared', 'remove-shared'])
    parser.add_argument('--fixture', required=True,
                        choices=['logger', 'config', 'both'])
    parser.add_argument('--dry-run', action='store_true',
                        help='print the would-be diff; write nothing')
    return parser.parse_args(argv)


def main(argv=None):
    args = parse_args(argv)
    if not args.classes.is_file():
        print('ERROR classes file not found: %s' % args.classes)
        return 2
    test_root = args.test_root.resolve()
    if not test_root.is_dir():
        print('ERROR test root not found: %s' % test_root)
        return 2

    counts = {}
    for fqcn in read_classes(args.classes):
        try:
            status, detail = process_class(fqcn, args, test_root)
        except SkipFile as reason:
            status, detail = 'SKIP', '%s -> %s' % (reason, test_file_for(test_root, fqcn))
        counts[status] = counts.get(status, 0) + 1
        print('[%s] %s %s' % (status, fqcn, detail))

    print('summary: ' + ', '.join('%s=%d' % kv for kv in sorted(counts.items())))
    return 0


if __name__ == '__main__':
    sys.exit(main())
